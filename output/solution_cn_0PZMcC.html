【C++】一种更快更妙的递归方法

#define LL long long
class Solution {
public:
    bool key=true;
    vector<LL> maxi;
    vector<LL> mini;
    void dfs(vector<int>& postorder,int s,int e){
        if(s>e||!key)return ;
        if(maxi[e]!=LONG_LONG_MAX)return ;
        if(s==e){
            maxi[e]=postorder[e];
            mini[e]=postorder[e];
            return ;
        }
        if(postorder[e-1]<postorder[e]){
            dfs(postorder,s,e-1);
            if(postorder[e]<maxi[e-1]){
                key=false;
                return;
            }
            maxi[e]=postorder[e];
            mini[e]=mini[e-1];
        }else if(postorder[s]>postorder[e]){
            dfs(postorder,s,e-1);
            if(postorder[e]>mini[s]){
                key=false;
                return;
            }
            maxi[e]=maxi[e-1];
            mini[e]=postorder[e];
        }else{
            int pos=lower_bound(postorder.begin()+s,postorder.begin()+e,postorder[e])-postorder.begin();
            dfs(postorder,s,pos-1);
            dfs(postorder,pos,e-1);
            if(maxi[pos-1]>postorder[e]||mini[e-1]<postorder[e]){
                key=false;
                return;
            }
            maxi[e]=maxi[e-1];
            mini[e]=mini[pos-1];
        }

        return;
    }
    bool verifyPostorder(vector<int>& postorder) {
        int size=postorder.size();
        if(size<=1){
            return true;
        }
        maxi.resize(size,LONG_LONG_MAX);
        mini.resize(size,LONG_LONG_MAX);
        dfs(postorder,0,size-1);
        return key;
    }   
};

本方法先假设其树形结构为二叉搜索树，每一层递归以数组末端为父节点，再使用二分查找确定数组中的子树分界处，利用记忆化搜索dfs两棵子树的区间，数组maxi、mini分别存储每一棵子树的最大值与最小值。
每一层递归判断是否满足条件: 左子树最大值 < 当前节点的值 < 右子树最小值
- 不满足则令全局变量为false以实现剪枝
- 满足则存储以当前节点为根的子树的最值：最大值=右子树最大值、最小值=左子树最小值
- 递归终止条件为遍历到叶子节点或key==false

这里时间复杂度的表达式较为复杂，在此不表，大概时间复杂度介于O(N)和O(N*log(N))之间。

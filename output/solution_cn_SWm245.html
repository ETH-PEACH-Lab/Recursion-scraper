Problem: 剑指 Offer 24. 反转链表

一：直接新建反转链表（搞笑方法）

```C# []
/*
* Definition for singly-linked list.
* public class ListNode {
* public int val;
* public ListNode next;
* public ListNode(int x) { val = x; }
* }
/
public class Solution {
public ListNode ReverseList(ListNode head) {
if(head == null || head.next == null) {
return head;
}
int length = 0;
ListNode pre = head;
while(pre != null) {
length++;
pre = pre.next;
}

    int[] values = new int[length];
    for(int i = length - 1; i >= 0; i--) {
        values[i] = head.val;
        head = head.next;
    }

    ListNode dummyHead = new ListNode(values[0]);
    ListNode res = dummyHead;
    for(int i = 1; i < length; i++){
        res.next = new ListNode(values[i]);
        res = res.next;
    }
    return dummyHead;
}

}

## 二：从前往后迭代

###### 过程写在了注释上。
```C# []
public class Solution {
    public ListNode ReverseList(ListNode head) {  
        //从后往前连接链表 ，举例
        //如何转移：用两个指针，p和q，默认开始为null
        //① head->1->2->3->4->5->Null   p->null, q->null
        //② p指向head的位置, p->1->..., head往后移head->2->...
        //  p.next = q = null, q再指向p: p->1->null, q->1->null, 
        //③ 指向head的位置, p->2->..., head往后移head->3->...
        //  p.next = q = 1->null, q再指向p: p->2->..., q->2->..., 
        //④ 指向head的位置, p->3->..., head往后移head->4->...
        //  p.next = q = 2->..., q再指向p: p->3->..., q->3->..., 
        //⑤ 指向head的位置, p->4->..., head往后移head->5->null
        //  p.next = q = 3->..., q再指向p: p->4->..., q->4->..., 
        //⑥ 指向head的位置, p->5->null, head往后移head->nul
        //  p.next = q = 4->..., q再指向p: p->5->..., q->5->..., 
        //⑦ head为null，结束循环。
        ListNode p = null, q = null;
        while(head != null) {
            p = head;
            head = head.next;
            p.next = q;
            q = p;
        }
        return p;
    }
}

```C# []
public class Solution {
public ListNode ReverseList(ListNode head) {
if(head == null || head.next == null) {
return head;
}
//双指针迭代
ListNode left = head, right = head.next;
left.next = null;
while(right != null) {
ListNode temp = right.next;
right.next = left;
left = right;
right = temp;
}
return left;
}
}

            head = head->next;
            left->next = right;
            right = left;
        }
        return right;
    }

## 三：递归
```C# []
public class Solution {
    public ListNode ReverseList(ListNode head) {  
        //按照迭代的规律，可以写出递归
        return Reverse(head, null);
    }
    //输入头节点和尾节点
    public ListNode Reverse(ListNode head, ListNode p) {
        if(head == null) {
            return p;  //当头节点为空，结束递归，返回尾节点
        }
        ListNode q = Reverse(head.next, head);
        head.next = p;  //用于调转的重点
        return q;
    }
}

```C++ []
/
* Definition for singly-linked list.
* struct ListNode {
* int val;
* ListNode next;
* ListNode(int x) : val(x), next(NULL) {}
* };
/
class Solution {
public:
ListNode reverseList(ListNode head) {
return reverse(head, NULL);
}
ListNode reverse(ListNode head, ListNode p) {
if(!head) {
return p; //返回了最后一个数
}
ListNode q = reverse(head->next, head);
head->next = p;
return q;
}
};

```Python3 []
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, x):
#         self.val = x
#         self.next = None

class Solution:
    def reverseList(self, head: ListNode) -> ListNode:
        def reverse(head: ListNode, p: ListNode) -> ListNode:
            if head == None:
                return p
            q, head.next = reverse(head.next, head), p
            return q
        return reverse(head, None)

Lets Take an example first and see its solution
s = 2[ab]2[d3[a2[c]]]



So as per the images, arrows represent returning a control, therefore we will recursion to solve this problem.
The Idea is to add all the sets of independent brackets
For every call, our current result is an empty string, we advance in the string and perform the following tasks

If the current character is an opening bracket, we hold our current result and recursively check for any nested brackets

To solve the current pair of brackets, we have to ensure all the brackets inside it are solved!!
After A recursive call is finished, we set the multiplier (integer that tells us how many times our string will be repeated) to 0
We take the control to the next index which has to been traversed yet,

Take an array and initialise all values to 0
Whenever an index is encountered, set it to 1 in the array.
0 ---> Untraversed, 1 ---> Traversed
Therefore we take the control to the next index with value 0

If current character is a letter, we add it to our current result and continue to advance in the input string
If the current character is an closing bracket,
We repeat our current result, multiplier times, and store it in a temporary string
We return the temporary string, this operation ends one set of brackets

Otherwise the current character is a digit,
It will give us the multiplier, so we will store it in another variable
This will help us to pass on the multiplier to another recursive call, if needed, after converting it into integer

Finally we will return the Current Result

Below is the code for the above mentioned approach

class Solution
{ // Runtime: 0 ms, faster than 100.00% of C++ online submissions for Decode String.
// Memory Usage: 6.4 MB, less than 78.36% of C++ online submissions for Decode String.
public:
int l, br = 0;
// Since string length can at worst be 30, create an array of length 30 to track the indices that have been traversed
int v[30]; 

string get_it(string s, int index, int mul) //(Original String, index to start with, multiplier)
{
    //Res to store immidiate results, 
    //result_so_far to get the result so far, 
    //num to get the multiplier
    string res = "", result_so_far = "", num = "";       
    int b = 0, i = index;
    while(i < l)
    {
        //Mark the index as traversed
        v[i] = 1;
        //If there is an opening
        if (s[i] == '[')                        
        {
            //For Every beginning, make a recursive call, as there might be nested Coded Strings like 3[a2[c]]
            //Store the result of the recursive call in result_so_far
            result_so_far += get_it(s, i + 1, stoi(num));
            //Set num as "", empty!
            num = "";
            //Take i to the next untraversed index
            while (v[i] != 0)
                i++;
        }
        //If there is an alphabet, add it to result_so_far
        else if (isalpha(s[i]))
        {
            result_so_far += s[i++];
        }
        //If its a closing bracket
        else if (s[i] == ']')
        {
            //Time to add the strings, m times,
            //This step is responsible for decoding(building up) the string
            for (int i = 1; i <= mul; i++)
                res += result_so_far;       //Immediate immresult
            return res;
        }
        //Otherwise the current character is a part of multiplier, i.e. number
        else
            num += s[i++];
    }   
    return result_so_far;
}
string decodeString(string s)
{
    l = s.size();
    //Set the l(size) places in the array v, as 0, stating all are untraversed
    for (int i = 0; i < l; i++)  
        v[i] = 0;

    return get_it(s, 0, 1);
}

};

class Solution {
    int l;
    int[] v = new int[30];

    // Recursive method to decode the string
    String getDecodedString(String s, int index, int mul) {
        String res = "", resultSoFar = "", num = "";
        int i = index;
        while (i < l) {
            v[i] = 1;
            if (s.charAt(i) == '[') {
                resultSoFar += getDecodedString(s, i + 1, Integer.parseInt(num));
                num = "";
                while (v[i] != 0) i++;
            } else if (Character.isLetter(s.charAt(i))) {
                resultSoFar += s.charAt(i++);
            } else if (s.charAt(i) == ']') {
                for (int j = 1; j <= mul; j++) {
                    res += resultSoFar;
                }
                return res;
            } else {
                num += s.charAt(i++);
            }
        }
        return resultSoFar;
    }

    // Main method to decode the given string
    public String decodeString(String s) {
        l = s.length();
        for (int i = 0; i < l; i++) {
            v[i] = 0;
        }
        return getDecodedString(s, 0, 1);
    }
}

class Solution:
def init(self):
self.l = 0
self.v = [0] * 30

# Recursive method to decode the string
def get_decoded_string(self, s: str, index: int, mul: int) -> str:
    res = ""
    result_so_far = ""
    num = ""
    i = index
    while i < self.l:
        self.v[i] = 1
        if s[i] == '[':
            result_so_far += self.get_decoded_string(s, i + 1, int(num))
            num = ""
            while self.v[i] != 0:
                i += 1
        elif s[i].isalpha():
            result_so_far += s[i]
            i += 1
        elif s[i] == ']':
            for j in range(1, mul + 1):
                res += result_so_far
            return res
        else:
            num += s[i]
            i += 1
    return result_so_far

# Main method to decode the given string
def decode_string(self, s: str) -> str:
    self.l = len(s)
    for i in range(self.l):
        self.v[i] = 0
    return self.get_decoded_string(s, 0, 1)

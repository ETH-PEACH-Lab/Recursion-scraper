3种思路，不得不说思路3看起来更优雅，大神的思路就是更加清晰。

思路1：设立虚拟头结点，借助辅助数据结构栈，空间复杂度是O（k）的

代码：

/**
 * Definition for a singly-linked list.
 * class ListNode {
 *     public $val = 0;
 *     public $next = null;
 *     function __construct($val = 0, $next = null) {
 *         $this->val = $val;
 *         $this->next = $next;
 *     }
 * }
 */
class Solution {

    /**
     * @param ListNode $head
     * @param Integer $k
     * @return ListNode
     */
    function reverseKGroup($head, $k) {
        if($k==1){//特殊情况
            return $head;
        }
        $dummyHead=new ListNode(null);//虚拟头结点
        $dummyHead->next=$head;
        $p=$dummyHead;//前一个节点
        $stack=new SplStack();
        $i=0;
        while($head!=null){
            $stack->push($head);//元素入栈
            $head=$head->next;//移动当前元素
            $i++;//标志+1
            if($i==$k){//标志位等于k，进行翻转
                list($h,$e)=$this->reverse($stack);
                $p->next=$h;//前一个节点的next指向新的头结点
                $e->next=$head;//尾节点指向下一个元素
                $p=$e;//前一个节点指向尾节点
                $i=0;//还原标志位
                $stack=new SplStack();//还原栈
            }
        }

        return $dummyHead->next;
    }

    //对栈里的元素进行翻转，返回翻转后的头结点和尾节点
    function reverse($stack){
        $node=null;
        while(!$stack->isEmpty()){
            if($node==null){
                $node=$stack->pop();
                $head=$node;
            }else{
                $node->next=$stack->pop();
                $node=$node->next;
            }
        }

        return [$head,$node];
    }
}

思路2：和思路1解法一样，不同点就是翻转的时候，不借助栈。空间复杂度会好很多。

代码：

/**
 * Definition for a singly-linked list.
 * class ListNode {
 *     public $val = 0;
 *     public $next = null;
 *     function __construct($val = 0, $next = null) {
 *         $this->val = $val;
 *         $this->next = $next;
 *     }
 * }
 */
class Solution {

    /**
     * @param ListNode $head
     * @param Integer $k
     * @return ListNode
     */
    function reverseKGroup($head, $k) {
        if($k==1){//特殊情况
            return $head;
        }
        $dummyHead=new ListNode(null);//虚拟头结点
        $dummyHead->next=$head;
        $p=$dummyHead;//考察的前一个节点
        $i=0;//标志位
        while($head!=null){
            $i++;
            $node=$head->next;//这里要提前保存一下head的next指针，因为在翻转链表的时候，head的next指向会改变，所以要提前保存
            if($i==$k){                
                list($h,$e)=$this->reverse($p->next,$head);//翻转链表
                $p->next=$h;   //设置前一个节点的next指针
                $e->next=$node;        
                $p=$e;                
                $i=0;//恢复标志位
            }
            $head=$node;
        }

        return $dummyHead->next;
    }

    //翻转链表，返回翻转后的头结点和尾节点
    function reverse($start,$end){        
        $cur=$start;
        $prev=null;
        while($cur!==$end){
            $next=$cur->next;
            $cur->next=$prev;
            $prev=$cur;
            $cur=$next;            
        }
        $end->next=$prev;

        return [$end,$start];
    }
}

思路3：递归和迭代结合，超经典，打卡labuladong

代码

/**
 * Definition for a singly-linked list.
 * class ListNode {
 *     public $val = 0;
 *     public $next = null;
 *     function __construct($val = 0, $next = null) {
 *         $this->val = $val;
 *         $this->next = $next;
 *     }
 * }
 */
class Solution {

    /**
     * @param ListNode $head
     * @param Integer $k
     * @return ListNode
     */
    function reverseKGroup($head, $k) {//递归翻转
        if($head==null){
            return null;
        }
        $a=$b=$head;
        for($i=0;$i<$k;$i++){//判断是否足够k个
            if($b==null){//不足直接返回
                return $head;
            }
            $b=$b->next;
        }
        $newHead=$this->reverse($a,$b);//得到新head
        $a->next=$this->reverseKGroup($b,$k);//递归翻转剩余链表

        return $newHead;
    }

    function reverse($a,$b){//[a,b)之间的链表
        $prev=null;
        $cur=$a;
        $next=$a;
        while($cur!==$b){//迭代翻转
            $next=$cur->next;;
            $cur->next=$prev;
            $prev=$cur;
            $cur=$next;
        }

        return $prev;
    }
}

Recursive without cache(Exceed Time Limit)

class Solution(object):
    def isMatchRecursive(self, s, p):
        # two exit condition of recursive
        # both are empty string
        if not s and not p:
            return True
        # s is not empty but p is empty
        if s and not p:
            return False
        # check if current characters are match
        cur_char_match = (s and s[0] == p[0]) or p[0] == '.'
        # split according to cur_char_match
        if cur_char_match:  # two are match
            if p[1:] and p[1] == '*':  # next character is '*'
                # repeat zero count
                zero_match = self.isMatchRecursive(s, p[2:])
                if zero_match:
                    return zero_match
                # repeat at least one time
                if s:  # if s is not empty
                    return self.isMatchRecursive(s[1:], p)
            else:  # next character is not '*'
                if s:
                    return self.isMatchRecursive(s[1:], p[1:])
        # two are not match
        if p[1:] and p[1] == '*':  # next character is '*'
            return self.isMatchRecursive(s, p[2:])

        # other condition all set to False
        return False


    def isMatch(self, s, p):
        """
        :type s: str
        :type p: str
        :rtype: bool
        """
        return self.isMatchRecursive(s, p)

Recursive with cache from top to down(Exceed Time Limit)

class Solution(object):

    def isMatchRecursive(self, s_idx, p_idx, s, p, matrix):
        # if s[s_Idx:] and p[p_idx:] is already cached in matrix[s_idx][p_idx], return it
        if matrix[s_idx][p_idx]:
            return matrix[s_idx][p_idx]
        # two exit condition of recursive
        # both s and p are iterated to the end
        if s_idx == len(s) and p_idx == len(p):
            matrix[s_idx][p_idx] = True
            return matrix[s_idx][p_idx]
        # s is not iterated to the end, but p is iterated to the end
        if s_idx < len(s) and p_idx == len(p):
            matrix[s_idx][p_idx] = False
            return matrix[s_idx][p_idx]
        # check if current characters are match
        cur_char_match = (s_idx < len(s) and (s[s_idx] == p[p_idx])) or p[p_idx] == '.'
        # split according to cur_char_match
        if cur_char_match:  # two are match
            if p_idx + 1 < len(p) and p[p_idx + 1] == '*':  # next character is '*'
                # repeat zero time
                zero_match = self.isMatchRecursive(s_idx, p_idx + 2, s, p, matrix)
                if zero_match:  # repeat zero time
                    matrix[s_idx][p_idx] = zero_match
                    return matrix[s_idx][p_idx]
                else:  # repeat at least one time
                    if s_idx < len(s):
                        matrix[s_idx][p_idx] = self.isMatchRecursive(s_idx + 1, p_idx, s, p, matrix)
                        return matrix[s_idx][p_idx]
            else:  # next character is not '*'
                if s_idx < len(s):
                    matrix[s_idx][p_idx] = self.isMatchRecursive(s_idx + 1, p_idx + 1, s, p, matrix)
                    return matrix[s_idx][p_idx]

        # two are not match
        if p_idx + 1 < len(p) and p[p_idx + 1] == '*':
            matrix[s_idx][p_idx] = self.isMatchRecursive(s_idx, p_idx + 2, s, p, matrix)
            return matrix[s_idx][p_idx]

        matrix[s_idx][p_idx] = False
        return matrix[s_idx][p_idx]




    def isMatch(self, s, p):
        """
        :type s: str
        :type p: str
        :rtype: bool
        """
        # create two dim matrix to cache
        matrix = [[0 for col in range(len(p) + 1)] for row in range(len(s) + 1)]
        return self.isMatchRecursive(0, 0, s, p, matrix)

Dynamic Programming from down to top, C code

bool createMatrixAndSearch(char* s, char* p, int sLen, int pLen) {
    // finally return result
    bool res = false;
    // create two dim matrix
    bool** matrix = (bool **) malloc(sizeof(bool *) * (sLen + 1));
    for (int row = sLen; row >= 0; row--) {
        matrix[row] = (bool *) malloc(sizeof(bool) * (pLen + 1));
        for (int col = pLen; col >= 0; col--) {
            // right-bottom most node is set to true, since two empty string are match
            if (row == sLen && col == pLen) {
                matrix[row][col] = true;
                continue;
            }
            // column of pLen is all set to false, except right-bottom most one, since pattern is empty
            if (row < sLen && col == pLen) {
                matrix[row][col] = false;
                continue;
            }
            // check if current character are match
            bool curCharMatch = (s[row] == p[col] || p[col] == '.');
            // split according to curCharMatch
            if (curCharMatch) { // if current character are match
                if (col + 1 < pLen && p[col + 1] == '*') { // if next character is '*'
                    // zero count or at least one time appearance
                    matrix[row][col] = matrix[row][col + 2] || (row < sLen && matrix[row + 1][col]) || false;
                } else { // next character is not '*'
                    matrix[row][col] = (row < sLen && matrix[row + 1][col + 1]) || false;
                }
            } else { // if current character are not match
                if (col + 1 < pLen && p[col + 1] == '*') { // if next character is '*'
                    // only possible when zero appearance of character right before '*'
                    matrix[row][col] = matrix[row][col + 2] || false;
                } else { // if next character is not '*'
                    matrix[row][col] = false;
                }
            }
        }
    }
    res = matrix[0][0];
    free(matrix);
    return res;
}
bool isMatch(char * s, char * p) {
    // get length of s
    char* ptr = s;
    int sLen = 0;
    while (*ptr != '\\0') {
        ptr++;
        sLen++;
    }
    // get length of p
    ptr = p;
    int pLen = 0;
    while (*ptr != '\\0') {
        ptr++;
        pLen++;
    }
    // create matrix and search in it
    return createMatrixAndSearch(s, p, sLen, pLen);
}

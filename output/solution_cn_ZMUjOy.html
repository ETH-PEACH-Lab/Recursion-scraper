解题思路

解法一：栈

代码

/**
     * 栈解法
     *
     * @param s
     * @return
     */
    public String decodeString(String s) {
        //保存左括号前的数字
        Stack<Integer> numStack = new Stack<>();
        //保存左括号前已扫描过的字母
        Stack<String> strStack = new Stack<>();

        StringBuilder result = new StringBuilder();
        int num = 0;
        //遍历所有的字符
        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);
            if (ch <= 'z' && ch >= 'a') {
                result.append(ch);
            } else if (Character.isDigit(ch)) {
                num = num * 10 + (ch - '0');
            } else if (ch == '[') {
                strStack.push(result.toString());
                result = new StringBuilder();

                numStack.push(num);
                num = 0;
            } else if (ch == ']') {
                StringBuilder tmp = new StringBuilder();
                int repeat = numStack.pop();

                while (repeat > 0) {
                    tmp.append(result);
                    repeat--;
                }

                result = new StringBuilder();
                if (!strStack.isEmpty()) {
                    result.append(strStack.pop()).append(tmp);
                } else {
                    result.append(tmp);
                }
            }
        }

        return result.toString();
    }

解法二：递归
定义一个递归函数 String decodeString(String s, int start, int end) 含义是将字符串s位于区间[start, end]之间的子串解码
该函数的处理过程：
1. 定义一个StringBuilder保存解码结果
2. 遍历区间[start, end]的字符：
- 如果遇到的是小写字母，直接添加到结果中，移动下标i
- 如果遇到的是数字，则向后遍历获取数字后第一个左括号和其对应的右括号下标，需要注意例子s1=300[ab[cd]]或s2=300、数字后不一定是左括号，可能还是数字
- 获得左括号下标后，[digitIndex, leftBracketInde)之间的字符串转化成数字，就是子串的重复次数repeat
- 递归将[]之前的字符串解码
- 将子串重复repeat次添加到结果中
- i = rightBracketIndex + 1 继续处理后序字符
3. 遇到数字后调用getBracketIndex方法返回数字后第一个左括号和与其匹配的右括号的下标，找到与当前左括号对应的右括号下标，遇到右括号且当前只有一个左括号或者i > end结束

代码

class Solution {
    public String decodeString(String s) {
        return decodeString(s, 0, s.length() - 1);
    }

    /**
     * 将字符串s位于区间[start, end]之间的子串解码
     *
     * @param s
     * @param start
     * @param end
     * @return
     */
    public String decodeString(String s, int start, int end) {
        if (start > end) {
            return "";
        }

        StringBuilder stringBuilder = new StringBuilder();

        int i = start;
        while (i <= end) {
            char ch = s.charAt(i);
            //如果是小写字母
            if (ch <= 'z' && ch >= 'a') {
                stringBuilder.append(ch);
                i++;
            } else if (Character.isDigit(ch)) {
                //记录数字起始下标
                int digitIndex = i;

                //获取数字后第一个左括号和其对应的右括号下标
                int[] indexPair = getBracketIndex(s, digitIndex, end);

                if (indexPair[0] > end) {
                    //整个串就是数字，如100，返回100
                    if (digitIndex == start) {
                        return s;
                    } else {
                        //整个串就是字符+数字，如ab100，返回ab
                        return s.substring(start, digitIndex);
                    }
                }

                int leftBracketIndex = indexPair[0];
                int rightBracketIndex = indexPair[1];

                //子串重复次数
                int repeat = stringToNumber(s.substring(digitIndex, leftBracketIndex));

                //递归将[]之前的字符串解码
                String decodeStr = decodeString(s, leftBracketIndex + 1, rightBracketIndex - 1);

                //子串重复repeat次添加到结果中
                while (repeat > 0) {
                    stringBuilder.append(decodeStr);
                    repeat--;
                }
                //继续处理后序字符
                i = rightBracketIndex + 1;
            }
        }

        return stringBuilder.toString();
    }

    /**
     * 遇到数字后返回数字后第一个左括号和与其匹配的右括号的下标，例如s1=300[ab[cd]]或s2=300
     *
     * @param s
     * @param start
     * @param end
     * @return
     */
    int[] getBracketIndex(String s, int start, int end) {
        //包括左括号和右括号的下标
        int[] result = new int[2];

        int i = start;
        int leftIndex, rightIndex;
        int leftBracketCount;

        //遍历查找左括号
        while (i <= end && s.charAt(i) != '[') {
            i++;
        }

        //到结尾也没有查找到'['
        if (i > end) {
            result[0] = i;
            return result;
        }

        //左括号的位置
        leftIndex = i;

        //记录未匹配的左括号数量
        leftBracketCount = 1;

        //从左括号下标leftIndex右侧开始查询与当前左括号匹配的右括号的下标
        i = leftIndex + 1;

        //找到与当前左括号对应的右括号下标，遇到右括号且当前只有一个左括号或者i > end结束
        while (i <= end && !(leftBracketCount == 1 && s.charAt(i) == ']')) {
            if (s.charAt(i) == '[') {
                leftBracketCount++;
            } else if (s.charAt(i) == ']') {
                //如果左括号数量不是1，且遇到右括号，将左括号数量减1
                leftBracketCount--;
            }
            i++;
        }

        //右括号的位置
        rightIndex = i;
        result[0] = leftIndex;
        result[1] = rightIndex;
        return result;
    }

    /**
     * 字符串转数字
     *
     * @param s
     * @return
     */
    private int stringToNumber(String s) {
        int result = 0;

        for (int i = 0; i < s.length(); i++) {
            result = result * 10 + (s.charAt(i) - '0');
        }

        return result;
    }
}

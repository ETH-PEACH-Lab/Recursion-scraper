736. c++超100的递归解法
👤 ffreturn 👍 1 2021-08-12 🖼 ↗ 查看原 Solution

解题思路

递归的解法
1. 把问题拆分成更小的单元，然后递归方式求解并累计，最后就是结果
2. 如何拆分呢？
- 按照()单位来拆分，比如多个(())也作为一个
- add 和 mul 就是把两部分算在一起即可
- let 则是保存字符串到数值的映射
- 考虑有多层级，所以会用一个map数组去保存

代码


class Solution {
private:
    // 按照当前层级记录的字符串到数值的映射：用于let 是设置数值使用
    vector<unordered_map<string, int>> scopes;

    // 拆分字符串为更小的单元
    vector<string> Split(const string& str)
    {
        stringstream ss(str);
        string curr;
        vector<string> res;
        // 标志位去计算是否是一个完整()
        int cnt = 0;
        // 完整()的一部分
        string part = "";
        while(std::getline(ss, curr, ' '))
        {
            if (curr.empty())
            {
                continue;
            }
            // 坑：这里需要全部都查找才能拆分出来
            for (char c : curr)
            {
                if (c == '(')
                {
                    ++cnt;
                }
                else if (c == ')')
                {
                    --cnt;
                }
            }

            if (!part.empty())
            {
                part += " ";
            }
            part += curr;

            if (cnt == 0)
            {
                res.push_back(part);
                part = "";
            }
        }

        if (!part.empty())
        {
            res.push_back(part);
        }

        // cout << "split [" << str << "] to "<< endl;
        // for (string c : res)
        // {
        //     cout << c << endl;
        // }
        // cout << "=======" << endl;
        return res;
    }

    int helper(const string& expression) 
    {
        // 如果不是 ( 开头， 可能就是纯数字或者let
        if (expression[0] != '(')
        {
            // 纯数字
            if ((expression[0] >= '0' && expression[0] <= '9') || (expression[0] == '-'))
            {
                cout << "raw number " << expression << endl;;
                return stoi(expression);
            }
            // 去scopes从后往前找找能已存在的结果，直接返回
            for (int i = scopes.size()-1; i >= 0; --i)
            {
                if (scopes[i].find(expression) != scopes[i].end())
                {
                    // cout << "existed scopes " << expression << " : " << scopes[i][expression] << endl;;
                    return scopes[i][expression];
                }
            }
        }

        // 下面就是三种情况 add let 和 mult
        int start = expression[1] == 'm' ? 6 : 5;
        // cout << "substr " << start << " size " << expression.size()-start-1 << endl;
        vector<string> parts = Split(expression.substr(start, expression.size()-start-1));
        if (expression[1] == 'a')
        {
            return evaluate(parts[0]) + evaluate(parts[1]);
        }
        else if (expression[1] == 'm')
        {
            return evaluate(parts[0]) * evaluate(parts[1]);
        }
        else {
            // let
            for (int j = 1; j < parts.size(); j += 2)
            {
                scopes.back()[parts[j-1]] = evaluate(parts[j]);
                // cout << "let " << parts[j-1] << " to [" << scopes.back()[parts[j-1]] << "]" << endl;
            }
            return evaluate(parts.back());
        }
    }

public:
    Solution() {
        scopes.push_back(unordered_map<string, int>());
    }

    int evaluate(string expression) {
        int res = 0;
        cout << "evaluate " << expression << endl;
        scopes.push_back(unordered_map<string, int>());

        res = helper(expression);

        // 回溯，结束当前层级的记录
        scopes.pop_back();

        return res;
    }
};

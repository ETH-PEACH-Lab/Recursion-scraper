解题思路

解题过程中两种解题思路,都是递归的想法, 不过代码思维方式不一样

思维一:

链表补全, 两个链表, 一个做进位合数链表, 另一个只做合数链表. 也就是说, l1链表个数少于l2, 那么就做ListNode(0)补充, 与l2平齐, 发生进位也在这个l1上做.

思维二:

做详细判断, 分情况去判断当前需要做哪些操作,不需要做哪些操作, 不需要补充链表, 只需要针对当前的递归循环参数的listnode是不是有数据做判断

上面两种思路, 大致相同, 思维一使用了swift 的特性? 给定默认值, 不去考虑数据的完整想, 在发挥语言特性上要优于思维二, 但是思维二逻辑相对而言更容易读懂.

/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     public var val: Int
 *     public var next: ListNode?
 *     public init(_ val: Int) {
 *         self.val = val
 *         self.next = nil
 *     }
 * }
 */
class Solution {
    func addTwoNumbers(_ l1: ListNode?, _ l2: ListNode?) -> ListNode? {
        var result: ListNode?

    if l1 != nil || l2 != nil {

        var node_1 = l1
        var node_2 = l2

        var sum = (node_1?.val ?? 0) + (node_2?.val ?? 0)
        if sum >= 10 {
            sum -= 10;
            node_1 = node_1?.next ?? ListNode.init(0)
            node_1?.val += 1
        } else {
            node_1 = node_1?.next
        }

        node_2 = node_2?.next

        result = ListNode.init(sum, addTwoNumbers(node_1, node_2))
    }

    return result
    }
}

代码: 思维二

/**
 * Definition for singly-linked list.
 * public class ListNode {
 *     public var val: Int
 *     public var next: ListNode?
 *     public init(_ val: Int) {
 *         self.val = val
 *         self.next = nil
 *     }
 * }
 */
func addTwoNumbers(_ l1: ListNode?, _ l2: ListNode?) -> ListNode? {
    var l1_value = 0
    var l2_value = 0
    var l1_next: ListNode?
    var l2_next: ListNode?

    if let l_1 = l1  {
        l1_value = l_1.val
        l1_next = l_1.next
    }

    if let l_2 = l2 {
        l2_value = l_2.val
        l2_next = l_2.next
    }

    var currentCodeValue = l1_value + l2_value
    var nextNode: ListNode?

    if currentCodeValue >= 10 {
        currentCodeValue -= 10
        if let l_1_next = l1_next {
            l_1_next.val += 1
            nextNode = addTwoNumbers(l_1_next, l2_next)
        } else if let l_2_next = l2_next {
            l_2_next.val += 1
            nextNode = addTwoNumbers(l1_next, l_2_next)
        } else {
            nextNode = ListNode.init(1)
        }
    } else {
        if l1_next != nil || l2_next != nil {
            nextNode = addTwoNumbers(l1_next, l2_next)
        }
    }

    return ListNode.init(currentCodeValue, nextNode)
}

func createListNode(array: [Int], tap: Int, preListNode: ListNode?) -> ListNode? {
    if tap < 0 {
        return preListNode;
    }

    let node = ListNode.init(array[tap], preListNode)

    return createListNode(array: array, tap: tap - 1, preListNode: node)
}

func_1 打印链表

func printListNode(listNode: ListNode?) {
    guard let node = listNode else {
        print("node 为nil")
        return
    }

    var currentNode: ListNode? = node
    while (currentNode != nil) {
        print(currentNode!.val)
        currentNode = currentNode?.next
    }
}

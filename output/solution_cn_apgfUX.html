因为 Lisp 语法本身就是递归定义的，所以很自然会采用递归的解法。根据题意，容易写出语法：

Expr            -> ( Operation )
                 | integer
                 | name
Operation       -> let AssignmentList Expr
                 | add Expr Expr
                 | mult Expr Expr
AssignmentList  -> name Expr AssignmentList'
AssignmentList' -> name Expr AssignmentList'
                 | ε

不过上面的语法不是 LL(1) 的，而是 LL(2) 的。在代码实现中，处理某个产生式的时候，要前瞻两个符号才能确定下一个产生式的选择。在本题中代码并不难实现（本文最下面会贴代码）

要想尝试将 LL(2) 语法转换成 LL(1) 的话，首先要找到问题所在。
手工/程序自动化计算每一个产生式的 FIRST+ 集合后注意到：FIRST+ 集合冲突发在最后两个产生式，这两个产生式都属于同一个非终结符 AssignmentList'，且对应的 FIRST+ 集合中都含有 name。
如果程序执行到某处，选择了规则 AssignmentList -> name Expr AssignmentList'，在消耗了 name Expr 符号后，紧接着要处理左侧是 AssignmentList' 的产生式，它代表的意思是紧接 0 到多个 name Expr，如果当前前瞻一个符号是 name，程序不知道要选择产生式左侧是 AssignmentList' 中的哪一个规则。因为此时程序如果选择采用产生式 AssignmentList' -> ε 的话，程序此时已经消耗了产生式 AssignmentList -> name Expr AssignmentList' 中的所有符号和产生式 Operation -> let AssignmentList Expr 中的 let AssignmenList 这两个符号，而紧接着非终结符 Expr，它能代表 ( Operation )、integer、name 中的一种。这意味着，最后两个产生式对于前瞻符号 name，都是合法、可以「消耗」的。

对上述语法尝试采取「提取左因子」的方法进行转换，有可能在至多前瞻一个符号下，该语法是无回溯的。不过上面每个产生式的右侧看上去好像没有共同的「左因子」。但根据上面的分析，我们知道应该让产生式 Operation -> let AssignmentList Expr 中的符号 Expr 交由非终结符 AssignmentList 处理，所以先改写语法成这样子：

Expr            -> ( Operation )
                 | integer
                 | name
Operation       -> let LetOperandList
                 | add Expr Expr
                 | mult Expr Expr
LetOperandList  -> name Expr LetOperandList'
LetOperandList' -> name Expr LetOperandList'
                 | ( Operation )
                 | integer
                 | name

接着，注意到上面非终结符 LetOperandList' 产生式的右侧的左侧都有共同的「左因子」 name。为了清除该「左因子」，我们引入非终结符 LetOperandListNameRightAfter：

Expr            -> ( Operation )
                 | integer
                 | name
Operation       -> let LetOperandList
                 | add Expr Expr
                 | mult Expr Expr
LetOperandList  -> name Expr LetOperandList'
LetOperandList' -> name LetOperandNameRightAfter
                 | ( Operation )
                 | integer
LetOperandNameRightAfter -> Expr LetOperandList'
                          | ε

可以手工/程序自动化计算证明该语法是 LL(1) 的。

关于嵌套作用域，非常简单。
- 当进入一层「括号」时，就创建新的作用域。
- 当退出一层「括号」时，就销毁最内层的作用域。
- 要创建/更新某个变量的值的时候，只在最内层的作用域中新建/更新变量的值。
- 要查找某个变量名的值的时候，先从最内部作用域中查找该变量的值，如果找到就返回；如果没有找到，就到外面一层的作用域中查找该变量名的值，如果找到就返回；如果没有找到，就到再外面一层的作用域……

以上四种操作，在本题中都可以用非常简单的数据结构 vector<unordered_map<string, int>> 来实现。

LL(1) 语法实现的代码：

class Solution {
enum TokenType {
    TOKEN_TYPE_LEFT_PARENTHESIS,
    TOKEN_TYPE_RIGHT_PARENTHESIS,
    TOKEN_TYPE_LET,
    TOKEN_TYPE_ADD,
    TOKEN_TYPE_MULT,
    TOKEN_TYPE_NAME,
    TOKEN_TYPE_INTEGER,
    TOKEN_TYPE_EOF
};

public:
    int evaluate(string expression) {
        // 递归下降法
        // 分成三个模块：词法分析、语法分析、嵌套作用域

        str = &expression;
        next_token();
        return expr();
    }

private:
    int pos = 0;
    string *str;
    enum TokenType t;
    string name;
    int integer;
    // 作用域符号表列表，用于处理嵌套的作用域。第 i 个符号表用于第 i 层的作用域
    vector<unordered_map<string, int>> scopes;

    int expr() {
        int res;
        switch (t) {
        case TOKEN_TYPE_LEFT_PARENTHESIS:
            /* Expr -> ( Operation ) */
            next_token();
            scope_push();
            res = operation();
            scope_pop();
            next_token();
            break;
        case TOKEN_TYPE_INTEGER:
            /* Expr -> integer */
            res = integer;
            next_token();
            break;
        case TOKEN_TYPE_NAME:
            /* Expr -> name */
            res = scope_lookup(name);
            next_token();
            break;
        }
        return res;
    }

    int operation() {
        int res;
        switch (t) {
        case TOKEN_TYPE_LET:
            /* Operation -> let LetOperandList */
            next_token();
            res = let_operand_list();
            break;
        case TOKEN_TYPE_ADD:
            /* Operation -> add Expr Expr */
            next_token();
            res = expr() + expr();
            break;
        case TOKEN_TYPE_MULT:
            /* Operation -> mult Expr Expr */
            next_token();
            res = expr() * expr();
            break;
        }
        return res;
    }

    int let_operand_list() {
        /* LetOperandList -> name Expr LetOperandList' */
        int res;
        while (true) {
            switch (t) {
            case TOKEN_TYPE_LEFT_PARENTHESIS:
                /* LetOperandList' -> ( Operation ) */
                next_token();
                scope_push();
                res = operation();
                scope_pop();
                next_token();
                goto out;
            case TOKEN_TYPE_INTEGER:
                /* LetOperandList' -> integer */
                res = integer;
                next_token();
                goto out;
            case TOKEN_TYPE_NAME:
                /* LetOperandList' -> name LetOperandListNameRightAfter
                   LetOperandListNameRightAfter -> Expr LetOperandList'
                                                 | ε */
                string n = move(name);
                next_token();
                if (t == TOKEN_TYPE_RIGHT_PARENTHESIS) {
                    res = scope_lookup(n);
                    goto out;
                }
                int val = expr();
                scope_insert(n, val);
            }
        }
        out:
        return res;
    }

    /* 作用域辅助函数 */
    int scope_lookup(string &name) {
        int res;
        for (int i = scopes.size() - 1; i >= 0; i--) {
            if (scopes[i].count(name) == 1) {
                res = scopes[i][name];
                break;
            }
        }
        return res;
    }

    void scope_insert(string &name, int val) {
        scopes.back()[name] = val;
    }

    void scope_push() {
        scopes.emplace_back();
    }

    void scope_pop() {
        scopes.pop_back();
    }

    /* 词法分析 */
    void next_token() {
        while (pos < str->size() && (*str)[pos] == ' ') {
            pos++;
        }

        if (pos == str->size()) {
            t = TOKEN_TYPE_EOF;
            return;
        }

        char c = (*str)[pos++];
        bool is_negative = false;
        switch (c) {
        case '(': t = TOKEN_TYPE_LEFT_PARENTHESIS; break;
        case ')': t = TOKEN_TYPE_RIGHT_PARENTHESIS; break;
        case '-':
            is_negative = true;
            c = (*str)[pos++];
        case '0': case '1': case '2': case '3': case '4': case '5':
        case '6': case '7': case '8': case '9':
            integer = c - '0';
            while (pos < str->size() && isdigit((*str)[pos])) {
                integer = integer * 10 + ((*str)[pos++] - '0');
            }
            if (is_negative) {
                integer = -integer;
            }
            t = TOKEN_TYPE_INTEGER;
            break;
        default: /* 变量或操作符 */
            name = c;
            while (pos < str->size() && (isdigit((*str)[pos]) || isalpha((*str)[pos]))) {
                name += (*str)[pos++];
            }
            if (name == "let") {
                t = TOKEN_TYPE_LET;
            } else if (name == "add") {
                t = TOKEN_TYPE_ADD;
            } else if (name == "mult") {
                t = TOKEN_TYPE_MULT;
            } else {
                t = TOKEN_TYPE_NAME;
            }
        }
    }
};

class Solution {
enum TokenType {
    TOKEN_TYPE_LEFT_PARENTHESIS,
    TOKEN_TYPE_RIGHT_PARENTHESIS,
    TOKEN_TYPE_LET,
    TOKEN_TYPE_ADD,
    TOKEN_TYPE_MULT,
    TOKEN_TYPE_NAME,
    TOKEN_TYPE_INTEGER,
    TOKEN_TYPE_EOF
};

public:
    int evaluate(string expression) {
        // 递归下降法
        // 分成三个模块：词法分析、语法分析、嵌套作用域
        /* LL(2) 语法：
           起始符号 Expr
           Expr -> ( Operation )
                 | integer
                 | name
           Operation -> let AssignmentList Expr
                      | add Expr Expr
                      | mult Expr Expr
           AssignmentList -> name Expr Assignment'
           AssignmentList' -> name Expr AssignmentList'
                            | ε
        */
        // 其中，表达式：a，根据题意该表达式是合法的，但没有定义其值

        str = &expression;
        next_token();
        next_token();
        return expr();
    }

private:
    int pos = 0;
    string *str;
    enum TokenType t, t2; // 至多前瞻 2 个符号
    string name, name2;
    int integer, integer2;
    // 作用域符号表列表，用于处理嵌套的作用域。第 i 个符号表用于第 i 层的作用域
    vector<unordered_map<string, int>> scopes;

    int expr() {
        int res;
        switch (t) {
        case TOKEN_TYPE_LEFT_PARENTHESIS:
            /* Expr -> ( Operation ) */
            next_token();
            scope_push();
            res = operation();
            scope_pop();
            next_token();
            break;
        case TOKEN_TYPE_INTEGER:
            /* Expr -> integer */
            res = integer;
            next_token();
            break;
        case TOKEN_TYPE_NAME:
            /* Expr -> name */
            res = scope_lookup(name);
            next_token();
            break;
        }
        return res;
    }

    int operation() {
        int res;
        switch (t) {
        case TOKEN_TYPE_LET:
            /* Operation -> let AssignmentList Expr */
            next_token();
            assignment_list();
            res = expr();
            break;
        case TOKEN_TYPE_ADD:
            /* Operation -> add Expr Expr */
            next_token();
            res = expr() + expr();
            break;
        case TOKEN_TYPE_MULT:
            /* Operation -> mult Expr Expr */
            next_token();
            res = expr() * expr();
            break;
        }
        return res;
    }

    void assignment_list() {
        /* AssignmentList -> name Expr Assignment' */
        // 这里前瞻两个符号以确定到底用哪个产生式
        do {
            /* AssignmentList' -> name Expr AssignmentList'
                                | ε */
            string n = name;
            next_token();
            int val = expr();
            scope_insert(n, val);
        } while (t == TOKEN_TYPE_NAME && t2 != TOKEN_TYPE_RIGHT_PARENTHESIS);
    }

    /* 作用域辅助函数 */
    int scope_lookup(string &name) {
        int res;
        for (int i = scopes.size() - 1; i >= 0; i--) {
            if (scopes[i].count(name) == 1) {
                res = scopes[i][name];
                break;
            }
        }
        return res;
    }

    void scope_insert(string &name, int val) {
        scopes.back()[name] = val;
    }

    void scope_push() {
        scopes.emplace_back();
    }

    void scope_pop() {
        scopes.pop_back();
    }

    /* 词法分析 */
    void next_token() {
        while (pos < str->size() && (*str)[pos] == ' ') {
            pos++;
        }

        t = t2;
        name = move(name2);
        integer = integer2;

        if (pos == str->size()) {
            t2 = TOKEN_TYPE_EOF;
            return;
        }

        char c = (*str)[pos++];
        bool is_negative = false;
        switch (c) {
        case '(': t2 = TOKEN_TYPE_LEFT_PARENTHESIS; break;
        case ')': t2 = TOKEN_TYPE_RIGHT_PARENTHESIS; break;
        case '-':
            is_negative = true;
            c = (*str)[pos++];
        case '0': case '1': case '2': case '3': case '4': case '5':
        case '6': case '7': case '8': case '9':
            integer2 = c - '0';
            while (pos < str->size() && isdigit((*str)[pos])) {
                integer2 = integer2 * 10 + ((*str)[pos++] - '0');
            }
            if (is_negative) {
                integer2 = -integer2;
            }
            t2 = TOKEN_TYPE_INTEGER;
            break;
        default: /* 变量或操作符 */
            name2 = c;
            while (pos < str->size() && (isdigit((*str)[pos]) || isalpha((*str)[pos]))) {
                name2 += (*str)[pos++];
            }
            if (name2 == "let") {
                t2 = TOKEN_TYPE_LET;
            } else if (name2 == "add") {
                t2 = TOKEN_TYPE_ADD;
            } else if (name2 == "mult") {
                t2 = TOKEN_TYPE_MULT;
            } else {
                t2 = TOKEN_TYPE_NAME;
            }
        }
    }
};

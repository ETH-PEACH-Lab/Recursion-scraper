Intuition

The problem involves checking if a given string s matches a pattern p where the pattern may include wildcard characters:

  • ? matches any single character.
  • ( * ) matches any sequence of characters (including an empty sequence).
    To solve this problem, we use dynamic programming to build up the solution based on subproblems. We consider various ways to match parts of the string with parts of the pattern using wildcard characters.

Approach 1: Recursion

  • This approach solves the problem by breaking it down into smaller subproblems, but it doesn't optimize for repeated calculations, which can lead to inefficiencies for large inputs.

Complexity

  • Time complexity: O(n*m)
  • Space complexity: O(n*m)

Code

```cpp []
class Solution {
private:
bool solve(string &s,string &p,int i,int j)
{
//base case
if(i==0 && j==0) return true;

    if(i>0 && j==0) return false;

    if(i==0 && j>0)
    {
        for(int k=1;k<=j;k++)
        {
            if(p[k-1]!='*') return false;
        }
        return true;
    }

        //match
        if(s[i-1]==p[j-1] || p[j-1]=='?') return solve(s,p,i-1,j-1);
        else if(p[j-1]=='*') 
            return ( solve(s,p,i-1,j)|| solve(s,p,i,j-1));
        else return false;
}

public:
bool isMatch(string s, string p) {
return solve(s,p,s.length(),p.length());
}
};

class Solution {
private:
    bool solveMem(string &s,string &p,int i,int j,vector<vector<int>> &dp)
    {
        //base case
        if(i==0 && j==0) return true;

        if(i>0 && j==0) return false;

        if(i==0 && j>0)
        {
            for(int k=1;k<=j;k++)
            {
                if(p[k-1]!='*') return false;
            }
            return true;
        }

        if(dp[i][j]!=-1) return dp[i][j];

        //match
        if(s[i-1]==p[j-1] || p[j-1]=='?')  return dp[i][j]=solveMem(s,p,i-1,j-1,dp);
        else if(p[j-1]=='*') 
            return dp[i][j]=( solveMem(s,p,i-1,j,dp)|| solveMem(s,p,i,j-1,dp));
        else return false;
    }    

public:
    bool isMatch(string s, string p) {

         int n=s.length();
         int m=p.length();

         vector<vector<int>> dp(n+1,vector<int> (m+1,-1));
         return solveMem(s,p,n,m,dp);

Approach 3: Tabulation

  • Create a 2D DP table where dp[i][j] represents if the first i characters of the string s match the first j characters of the pattern p.
  • Use this table to iteratively fill in values based on whether characters match or if the pattern contains wildcards (? or *).

Complexity

  • Time complexity: O(n*m)
  • Space complexity: O(n*m)

Code

```cpp []
class Solution {
private:
bool solveTab(string s,string p)
{
int n=s.length();
int m=p.length();
vector> dp(n+1,vector (m+1,0));
dp[0][0]=true;

    for(int j=1;j<=m;j++)
    {
        bool flag=true;
        for(int k=1;k<=j;k++)
        {
            if(p[k-1]!='*')
            {
                flag=false;
                break;
            }
        }
        dp[0][j]=flag;
    }

    for(int i=1;i<=n;i++)
    {
        for(int j=1;j<=m;j++)
        {
            //match
            if(s[i-1]==p[j-1] || p[j-1]=='?')  dp[i][j]=dp[i-1][j-1];
            else if(p[j-1]=='*') 
                 dp[i][j]= ( dp[i-1][j]|| dp[i][j-1]);
            else dp[i][j]=false;
        }
    }

    return dp[n][m];
}

public:
bool isMatch(string s, string p) {

     return solveTab(s,p);

}

};

class Solution {
private:
    bool solveSO(string s,string p)
    {
        int n=s.length();
        int m=p.length();
        vector<int> prev(m+1,0);
        vector<int> curr(m+1,0);

        prev[0]=true;

        for(int j=1;j<=m;j++)
        {
            bool flag=true;
            for(int k=1;k<=j;k++)
            {
                if(p[k-1]!='*')
                {
                    flag=false;
                    break;
                }
            }
            prev[j]=flag;
        }

        for(int i=1;i<=n;i++)
        {
            for(int j=1;j<=m;j++)
            {
                //match
                if(s[i-1]==p[j-1] || p[j-1]=='?')  curr[j]=prev[j-1];
                else if(p[j-1]=='*') 
                     curr[j]= ( prev[j]|| curr[j-1]);
                else curr[j]=false;
            }
            prev=curr;
        }

        return prev[m];
    }     
public:
    bool isMatch(string s, string p) {
        return solveSO(s,p);
    }
};

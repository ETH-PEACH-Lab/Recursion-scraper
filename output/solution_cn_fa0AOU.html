思路

无论递归还是dp，记录（返回）一个在当前数组中，自己比对方多拿的差值的最大值

解法1 记忆化递归

时间复杂度O(n)，缺点是递归栈的空间开销
memset()函数用于初始化数组，只能是0或-1，因为函数是在地址空间逐字节补充的，只能全赋值为0或1，头文件

class Solution {
public:
    int memo[21][21];
    bool PredictTheWinner(vector<int>& nums) {
        memset(memo, -1, sizeof(memo));
        if (total(nums, 0, nums.size()-1) >= 0) return true;
        return false;
    }
    int total(vector<int>& nums, int start, int end) {
        if (start == end) return nums[start];
        if (memo[start][end] != -1) return memo[start][end];
        int leftTotal = -total(nums, start+1, end) + nums[start];
        int rightTotal = -total(nums, start, end-1) + nums[end];
        memo[start][end] = max(leftTotal, rightTotal);
        return memo[start][end];
    }
};

解法2 dp

空间O(n^2)，可以压缩为O(n)

class Solution {
public:
    bool PredictTheWinner(vector<int>& nums) {
        int sz = nums.size();
        int dp[sz][sz];
        for (int i = 0; i < sz; i++) dp[i][i] = nums[i];
        for (int i = sz-2; i >= 0; i--) {
            for (int j = i+1; j < sz; j++) {
                dp[i][j] = max(nums[i] - dp[i+1][j], nums[j] - dp[i][j-1]);
            }
        }
        return dp[0][sz-1] >= 0;
    }
};

思路

首先，如果有1，2，3个石头，先手赢，但是如果有4个石头，先手就不可能赢。
那么，如果是更多的石头呢？
如果是5个石头，那么先手只要拿走1个，给后手造成4个石头的局面，那么先手就会赢，6，7也一样，而8，先手依然是必败。

代码

class Solution {
public:
    bool canWinNim(int n) {
        return n % 4 != 0;
    }

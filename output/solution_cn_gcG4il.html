解题思路

此处撰写解题思路

执行思路：
1、把最外层的括号去掉
2、提取出第一个单词，将单词作为后续要执行哪部分功能的标识
3、根据标识执行3个功能
3-1、赋值功能，由于要赋值的字符串是成对出现的，用2个变量key，val分别表示，根据key把val添加到哈希表对应的栈中，使用栈是因为每次取值都要取最新设置的val，这与栈先进后出的特性相符。赋值需要注意的是，如果val包含括号，则应该递归调用自身获取子串的计算结果，如果val不是数字则意味着val是以前赋过值的标量名，从哈希表中获取该变量的最新值，如果val是单纯的数字，就直接类型转换入栈即可。另外还要对入栈次数进行统计，退出时根据入栈次数把本层递归入栈的值全部弹出，保证后续调用的正确性，赋值功能的最后要根据字符串末尾部分返回结果，还是那3种情况，如果尾字符串包含括号就调用递归函数，返回递归结果，如果是变量名就返回字典中该变量的最新值，如果是数字，就直接返回。
3-2、相乘功能，只有乘数和被乘数2个需要记录的字符串，每个字符串同样分3种情况，和上面类似，包含括号就调递归函数，是变量就从哈希表中取最新值，是数字就直接参与乘法计算。
3-3、相加功能，整个过程和相乘完全一致，只是将最后的相乘运算变为相加即可。
4、逐位读取目标字符串，分割出每个单词，使用计数变量对左括号的数量计数，只有计数为0时才进行单词分割，其他情况就原样保存，大于0说明需要调用递归函数，当前字符串由下一层递归来处理。
5、将所有功能附加在一起返回最终处理结果

个人将近3个小时分析调试，写了180多行的冗余代码，思路仅供参考和启发

public class Solution {
    List<string> keywords = new List<string>(){"let","mult","add"};
    Dictionary<string,Stack<int>> val_Stc_Map = new Dictionary<string,Stack<int>>();
    public int Evaluate(string expression) {
        string expr = expression.Substring(1,expression.Length-2);
        int pos = 0;
        StringBuilder sb = new StringBuilder();
        while(pos < expr.Length && expr[pos] != ' ')
        {
            sb.Append(expr[pos]);
            pos++;
        }
        expr = expr.Substring(pos+1);
        pos = keywords.IndexOf(sb.ToString());
        switch(pos)
        {
            case 0:
                return SetValueOper(expr);
            case 1:
                return MultiplyOper(expr);
            case 2:
                return AddOper(expr);
        }
        //return val_Stc_Map["x"].Peek();
        return -1;
    }
    private int SetValueOper(string expr)
    {
        string key = "",val = "";
        StringBuilder sb = new StringBuilder();
        int leftCnt = 0;
        Dictionary<string,int> setCnt = new Dictionary<string,int>();
        for(int i = 0; i < expr.Length;i++)
        {
            if(expr[i] != ' ')
            {
                sb.Append(expr[i]);
                if(expr[i] == '(')
                    leftCnt++;
                else if(expr[i] == ')')
                    leftCnt--;
            }
            else
            {
                if(leftCnt > 0)
                {
                    sb.Append(expr[i]);
                    continue;
                }
                if(key == "")
                    key = sb.ToString();
                else
                    val = sb.ToString();
                if(key != "" && val != "")
                {
                    int mapVal = -1;
                    if(val.Contains('('))
                        mapVal = Evaluate(val);
                    else if(val_Stc_Map.ContainsKey(val))
                        mapVal = val_Stc_Map[val].Peek();
                    else
                        mapVal = int.Parse(val);
                    if(!val_Stc_Map.ContainsKey(key))
                    {
                        val_Stc_Map[key] = new Stack<int>();
                    }
                    if(!setCnt.ContainsKey(key))
                        setCnt[key] = 0;
                    val_Stc_Map[key].Push(mapVal);
                    setCnt[key]++;
                    key = "";
                    val = "";
                }
                sb.Clear();
            }
        }
        string endStr = sb.ToString();
        if(endStr.Contains('('))
        {
            int ret = Evaluate(endStr);
            foreach(var el in setCnt)
            {
                for(int i = 0; i < el.Value;i++)
                    val_Stc_Map[el.Key].Pop();
            }
            return ret;
        }
        else if(val_Stc_Map.ContainsKey(endStr))
        {
            int ret = val_Stc_Map[endStr].Peek();
            foreach(var el in setCnt)
            {
                for(int i = 0; i < el.Value;i++)
                    val_Stc_Map[el.Key].Pop();
            }
            return ret;
        }
        else
        {
            int ret = int.Parse(endStr);
            foreach(var el in setCnt)
            {
                for(int i = 0; i < el.Value;i++)
                    val_Stc_Map[el.Key].Pop();
            }
            return ret;
        }
    }
    private int MultiplyOper(string expr)
    {
        StringBuilder sb = new StringBuilder();
        int leftCnt = 0;
        int first = -1;
        for(int i = 0; i < expr.Length;i++)
        {
            if(expr[i] != ' ')
            {
                sb.Append(expr[i]);
                if(expr[i] == '(')
                    leftCnt++;
                else if(expr[i] == ')')
                    leftCnt--;
            }
            else
            {
                if(leftCnt > 0)
                {
                    sb.Append(expr[i]);
                    continue;
                }
                string str = sb.ToString();
                if(str.Contains('('))
                    first = Evaluate(str);
                else if(val_Stc_Map.ContainsKey(str))
                    first = val_Stc_Map[str].Peek();
                else
                    first = int.Parse(str);
                sb.Clear();
            }
        }
        string endstr = sb.ToString();
        if(endstr.Contains('('))
            return first * Evaluate(endstr);
        else if(val_Stc_Map.ContainsKey(endstr))
            return first * val_Stc_Map[endstr].Peek();
        else 
            return first * int.Parse(endstr);
    }
    private int AddOper(string expr)
    {
        StringBuilder sb = new StringBuilder();
        int leftCnt = 0;
        int first = -1;
        for(int i = 0; i < expr.Length;i++)
        {
            if(expr[i] != ' ')
            {
                sb.Append(expr[i]);
                if(expr[i] == '(')
                    leftCnt++;
                else if(expr[i] == ')')
                    leftCnt--;
            }
            else
            {
                if(leftCnt > 0)
                {
                    sb.Append(expr[i]);
                    continue;
                }
                string str = sb.ToString();
                if(str.Contains('('))
                    first = Evaluate(str);
                else if(val_Stc_Map.ContainsKey(str))
                    first = val_Stc_Map[str].Peek();
                else
                    first = int.Parse(str);
                sb.Clear();
            }
        }
        string endstr = sb.ToString();
        if(endstr.Contains('('))
            return first + Evaluate(endstr);
        else if(val_Stc_Map.ContainsKey(endstr))
            return first + val_Stc_Map[endstr].Peek();
        else
            return first + int.Parse(endstr);
    }
}

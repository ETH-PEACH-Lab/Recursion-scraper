 ```java []
class Solution {
    /**
     *递归求解
     * 返回  Sn 的 第 k 位字符 
     * @param n
     * @param k
     * @return
     */
    public char findKthBit(int n, int k) {
        if(n==1){
            return '0';
        }
        int mid=1<<(n-1);//2*(n-1)
        if(k==mid){ 
            return '1';
        }else if(k<mid){
            return findKthBit(n-1,k);//前一部分Si-1
        }
        return invertChar(findKthBit(n-1,(1<<n)-k));//后一部分reverse(invert(Si-1))
    }
    /**
     * 翻转字符
     * @param s
     * @return
     */
    private char invertChar(char s){
        if(s=='0'){
            return '1';
        }else{
            return '0';
        }
    }
}

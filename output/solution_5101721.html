Code

class Solution {
    public ListNode swapPairs(ListNode head) {
        if (head == null || head.next == null) return head;

        ListNode secondNode = head.next;
        head.next = secondNode.next;
        secondNode.next = head;
        head.next = swapPairs(head.next);

        return secondNode;
    }
}

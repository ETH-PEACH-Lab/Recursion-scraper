🚀 Quickest Solution Yet 🚀
👤 SumitNagrikar 👍 0 2024-10-28 🖼 ↗ 查看原 Solution

Intuition

To add two numbers represented by linked lists, we can leverage the fact that the digits are stored in reverse order, allowing us to add corresponding nodes directly from left to right. Since each node represents a digit, it’s similar to adding numbers manually: we add corresponding digits, manage a carry if needed, and move to the next digits.

Approach

  1. Initialize a Result List: We create a dummy head for the result linked list, which helps in building the final linked list and returning it easily.
  2. Define a Carry: Initialize carry to 0, which will store any overflow from summing two digits (if the sum is 10 or more).
  3. Loop Through the Lists:
    - Use a while loop to continue as long as either l1 or l2 has nodes remaining, or there is a carry.
    - At each step:
    • Initialize sum with the current carry.
    • If l1 is not null, add l1.val to sum and move l1 to its next node.
    • If l2 is not null, add l2.val to sum and move l2 to its next node.
    • Update carry to sum / 10 (integer division) and sum to sum % 10 (remainder, which is the current digit).
    • Create a new node with sum and attach it to the result list.
  4. Final Carry Check: If there is a leftover carry after the loop, create an additional node with the value of carry.
  5. Return the Result: Return result.next, which is the actual head of the newly formed list (skipping the dummy head).

Complexity

  • Time Complexity:

  • O(\max(m, n)), where m and n are the lengths of l1 and l2. We iterate through both lists once, so the time complexity depends on the longer list.

  • Space Complexity:

  • O(\max(m, n)), as we create a new list to store the result, which will have a length up to the maximum of m or n plus one if there’s a carry at the end.

/
* Definition for singly-linked list.
* function ListNode(val, next) {
* this.val = (val===undefined ? 0 : val)
* this.next = (next===undefined ? null : next)
* }
*/
/
* @param {ListNode} l1
* @param {ListNode} l2
* @return {ListNode}
*/
var addTwoNumbers = function(l1, l2) {
let result = new ListNode(0);//dummy node - starting point of our ans
let ptr = result;//pointing to result

let carry = 0;//every sum may or may not having carry

while(l1 !== null || l2 !== null){//we go inside if either of list having elements 
    let sum = carry;//first add carry to sum

    if(l1 !== null){//then we add l1 element
        sum += l1.val;
        l1 = l1.next;
    }

    if(l2 !== null){//then we add l2 element
        sum += l2.val;
        l2 = l2.next;
    }

    carry = Math.floor(sum / 10);//then find carry for next iteration
    sum = sum % 10;//find sum means remaining ele. for creating node

    ptr.next = new ListNode(sum);//creating node next to oor ptr
    ptr = ptr.next;//move pointer forward
}

if(carry === 1) {//At end, if carry remains, creat a node for it
    ptr.next = new ListNode(1);
}

return result.next;//returning the next node of result, cause result is dummy

};

class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def addTwoNumbers(l1, l2):
    result = ListNode(0)  # Dummy head node for the result list
    ptr = result  # Pointer to build the result list
    carry = 0  # Initialize carry to 0

    while l1 or l2 or carry:  # Continue if there are nodes or carry
        sum = carry  # Start with carry

        # Add l1's value if l1 is not None, then move to next node
        if l1:
            sum += l1.val
            l1 = l1.next

        # Add l2's value if l2 is not None, then move to next node
        if l2:
            sum += l2.val
            l2 = l2.next

        # Update carry and get current digit
        carry = sum // 10
        sum = sum % 10

        # Add sum as a new node to the result list
        ptr.next = ListNode(sum)
        ptr = ptr.next

    return result.next  # Return the head of the new list (skipping dummy node)

```java []
public class ListNode {
int val;
ListNode next;
ListNode() {}
ListNode(int val) { this.val = val; }
ListNode(int val, ListNode next) { this.val = val; this.next = next; }
}

public class AddTwoNumbers {

public ListNode addTwoNumbers(ListNode l1, ListNode l2) {
    ListNode result = new ListNode(0);  // Dummy head node
    ListNode ptr = result;  // Pointer to build the result list
    int carry = 0;  // Initialize carry to 0

    while (l1 != null || l2 != null || carry != 0) {  // Continue if there are nodes or carry
        int sum = carry;  // Start with carry

        // Add l1's value if l1 is not null, then move to next node
        if (l1 != null) {
            sum += l1.val;
            l1 = l1.next;
        }

        // Add l2's value if l2 is not null, then move to next node
        if (l2 != null) {
            sum += l2.val;
            l2 = l2.next;
        }

        // Update carry and get the current digit
        carry = sum / 10;
        sum = sum % 10;

        // Add sum as a new node to the result list
        ptr.next = new ListNode(sum);
        ptr = ptr.next;
    }

    return result.next;  // Return the head of the new list (skipping dummy node)
}

}
```

解法：递归算法；计算每一轮得分差值，如最终得分>=0，则获胜。

bool PredictTheWinner(int* nums, int numsSize){
    int n = numsSize;
    return dfs(nums, n, 0, n - 1) >= 0;
}

int dfs(int* nums, int n, int left, int right) {
    if (left == right) {
        return nums[left];
    }
    int pickLeft = nums[left] - dfs(nums, n, left + 1, right);
    int pickRight = nums[right] - dfs(nums, n, left, right - 1);
    return fmax(pickLeft, pickRight);
}

解法2：记忆化搜索，在解法1基础上有很多重复中间处理过程；
记忆中间结果，避免重复计算；

#define MAX(a, b) (((a) > (b)) ? (a) : (b))

int dfs(int* nums, int n, int left, int right, int **dp) {
    if (dp[left][right] != INT_MAX) {
        return dp[left][right];
    }
    if (left == right) {
        return nums[left];
    }

    int pickLeft = nums[left] - dfs(nums, n, left + 1, right, dp);
    int pickRight = nums[right] - dfs(nums, n, left, right - 1, dp);
    dp[left][right] = fmax(pickLeft, pickRight);
    return dp[left][right];
}

bool PredictTheWinner(int* nums, int numsSize){
    int n = numsSize;
    int **dp = (int **)malloc(n * sizeof(int *));
    for (int i = 0; i < n; i++) {
        dp[i] = (int *)malloc(n * sizeof(int));
        for (int j = 0; j < n; j++) {
            dp[i][j] = INT_MAX;
        }
    }
    return dfs(nums, n, 0, n - 1, dp) >= 0;
}

解法3：动态规划
由解法2直接改造成动态规划

bool PredictTheWinner(int* nums, int numsSize){
    int n = numsSize;
    int dp[n][n];
    memset(dp, 0, sizeof(dp));
    for (int i = 0; i < n; i++) {
        dp[i][i] = nums[i];
    }
    for (int i = n - 2; i >= 0; i--) {
        for (int j = i + 1; j < n; j++) {
            dp[i][j] = fmax(nums[i] - dp[i + 1][j], nums[j] - dp[i][j - 1]);
        }
    }
    return dp[0][n - 1] >= 0;
}

自己最初做的，算法正确，但超时，另外代码不够精简，算是学习路上的记忆吧。

typedef struct {
    int s1;
    int s2;
} SUM_S;

typedef struct {
    int cur;
    int left;
    int right;
    int s1;
    int s2;
} KEY_S;

typedef struct {
    KEY_S key;
    SUM_S ret;
    UT_hash_handle hh;
} HASHTABLE_S;

SUM_S dfs(int cur,HASHTABLE_S **ht, int* nums, int n, int left, int right, int s1, int s2) {
    HASHTABLE_S *tmp;
    KEY_S key;
    key.cur = cur;
    key.left = left;
    key.right = right;
    key.s1 = s1;
    key.s2 = s2;
    HASH_FIND(hh, *ht, &key, sizeof(KEY_S), tmp);
    if (tmp) {
        return tmp->ret;
    }

    SUM_S ret;
    if (left > right) {
        ret.s1 = s1;
        ret.s2 = s2;
        return ret;
    }
    int next = !cur;
    if (cur == 0) {
        SUM_S r1 = dfs(next, ht, nums, n, left + 1, right, s1 + nums[left], s2);
        SUM_S r2 = dfs(next, ht, nums, n, left, right - 1, s1 + nums[right], s2);
        ret = (r1.s1 > r2.s1) ? r1 : r2;
    } else {
        SUM_S r1 = dfs(next, ht, nums, n, left + 1, right, s1, s2 + nums[left]);
        SUM_S r2 = dfs(next, ht, nums, n, left, right - 1, s1, s2 + nums[right]);
        ret = (r1.s2 > r2.s2) ? r1 : r2;
    }

    tmp = (HASHTABLE_S *)malloc(sizeof(HASHTABLE_S));
    tmp->key = key;
    tmp->ret = ret;
    HASH_ADD(hh, *ht, key, sizeof(KEY_S), tmp);
    return ret;        
}

bool PredictTheWinner(int* nums, int numsSize){
    int n = numsSize;
    HASHTABLE_S *ht = NULL;
    HASHTABLE_S *node;
    HASHTABLE_S *tmp;
    SUM_S ret = dfs(0, &ht, nums, n, 0, n - 1, 0, 0);
    HASH_ITER(hh, ht, node, tmp) {
        HASH_DEL(ht, node);
        free(node);
    }
    return (ret.s1 >= ret.s2) ? true: false;
}

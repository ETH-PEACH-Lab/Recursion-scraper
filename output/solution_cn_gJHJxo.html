二 动态规划

2.1 动态规划

dp[i]表示F(i) 已知dp[0] = 0 dp[1] = 1 状态转换方程即递推方程 

var fib = function(N) {
    let dp = new Uint32Array(N + 1), i = 1
    dp[1] = 1
    while (i++ < N) dp[i] = dp[i - 1] + dp[i - 2]
    return N ? dp[N] : 0
};

2.2 滚动数组

dp[1]代替dp[i] 上轮dp[1]代替dp[i - 1] dp[0]代替dp[i - 2] 滚动：dp[0] = 上轮dp[1]
中间变量：暂存上轮dp[1]

var fib = function(N) {
    let dp = [0, 1], i = 1
    while (i++ < N) {
        const tmp = dp[1]
        dp[1] = dp[0] + dp[1]
        dp[0] = tmp
    }
    return N ? dp[1] : 0
};

运算顺序：先读取+号左边dp[0]，再赋值扣号内dp[0]，省去中间变量

var fib = function(N) {
    let dp = [0, 1], i = 1
    while (i++ < N) dp[1] = dp[0] + (dp[0] = dp[1])
    return N ? dp[1] : 0
};

2.3 状态机

用两个变量，分别代替dp[1]和dp[0]

var fib = function(N) {
    let s0 = 0, s1 = 1, i = 1
    while (i++ < N) s1 = s0 + (s0 = s1)
    return N ? s1 : 0
};

四 杨辉三角

杨辉三角左对齐后的对角线累加和即斐波那契数列。图来自《维基百科》

4.1 相邻

生成杨辉三角：中间位 = 上一行相邻元素和

var fib = function(N) {
    let M = generate(N), r = 0, i = N, j = -1
    while (i-- && ++j < M[i].length) r += M[i][j]
    return r
};
var generate = function(numRows) {
    return Array(numRows).fill().map((_, i, r) => r[i] = Array(i + 1).fill(1).map((v, j) => j > 0 && j < i ? r[i - 1][j - 1] + r[i - 1][j] : v))
};

4.2 对称性

生成杨辉三角：用对称性，从两端到中间，同时赋值j和i - j

var fib = function(N) {
    let M = generate(N), r = 0, i = N, j = -1
    while (i-- && ++j < M[i].length) r += M[i][j]
    return r
};
var generate = function(numRows) {
    for(var i = 0, r = new Array(numRows); i < numRows; i++) {
        r[i] = i < 10 ? new Uint8Array(i + 1) : i < 19 ? new Uint16Array(i + 1) : new Uint32Array(i + 1)
        r[i][0] = r[i][i] = 1
        for (var j = 1; j <= i >> 1; j++) 
            r[i][j] = r[i][i - j] = r[i - 1][j - 1] + r[i - 1][j]
    }
    return r
};

五 比内公式

var fib = function(N) {
    const s5 = Math.sqrt(5)
    return (Math.pow(.5 + s5 / 2, N) - Math.pow(.5 - s5 / 2, N)) / s5
};

六 矩阵快速幂

6.1 整数快速幂

const pow = (M, N) => {
    let r = 1
    while(N) {
        if (N & 1) r *= M
        M *= M
        N >>= 1
    }
    return r
}

6.2 矩阵快速幂

整数换成矩阵

const pow = (M, N) => {
    const r = [[1, 0], [0, 1]]
    while(N) {
        if (N & 1) r = multiply(r, M)
        M = multiply(M, M)
        N >>= 1
    }
    return r
}

矩阵乘法

const multiply = (M1, M2) => {
    const r = Array.from({length: M1.length}, _ => new Uint32Array(M2[0].length))
    for (let i = 0; i < M1.length; i++) // 遍历矩阵，水平扫描 缓存命中率 > 垂直扫描
        for (let j = 0; j < M1[0].length; j++)
            for (let k = 0; k < M2[0].length; k++) 
                r[i][k] += M1[i][j] * M2[j][k]
    return r
}


求解：M = [[1, 1], [1, 0]] F(N) = M的N - 1次方[0][0]项

var fib = function(N) {
    const M = [[1, 1], [1, 0]]
    return N ? pow(M, N - 1)[0][0] : 0
};

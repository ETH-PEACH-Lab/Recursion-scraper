class Solution {
public:
int num=0;
string str="";
string decodeString(string s) {
string out;
out=in(s);
return out;
}
//递归思路：每次递归先找前括号，找到前括号，分割字符串，
将前括号后面的 字符串提取，然后进入下一层，一直把所有前括号找完，
然后找后括号，找到返回，括号中间的字符串，乘以前面的系数，
这边用了一个全局变量暂存每一层更新的字符串。
string in(string s){
string out;
int one,two;
string ou;
one=s.find('[');
if(one!=-1){
ou=in(s.substr(one+1));
//更新字符串
if(str!="")s=s.substr(0,one+1)+str;
//统计括号前数字大小
int i=0;
int k=1;//计算数字
while((one-k>=0)&&s[one-k]>='0'&&s[one-k]<='9'){
i=i+(s[one-k]-'0')*pow(10, k-1);
k++;
}
while(i!=0){
out+=ou;
i--;
}
//更新当前字符串
s=s.substr(0,one-k+1)+out+s.substr(one+ou.size()+2);
//赋值给全局str
str=s;
}
two=s.find(']');
if(two!=-1){
num=two;
return s.substr(0,two);//返回中间字符串
}
return s;
}

};

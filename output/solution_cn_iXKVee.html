这个方法比较繁琐，实际可以精简很多。

按照题设，给出如下的语法：

init    : expr    // 测试样例不全是 let 开始
        ;
assign  : "let" (id expr)+ expr
        ;
addExpr : "add" expr expr
        ;
mulExpr : "mult" expr expr
        ;
base    : addExpr
        | mulExpr
        | assign
        ;
expr    : "(" base ")"
        | num
        | id
        ;

按照语法分析的方法，这里建立一个符号表。

这里建立的是命令式符号表。

  • 使用一个 unordered_map<std::string, std::int32_t> m_memory 保存当前环境。
  • 使用一个 std::vector<std::unordered_map<std::string, std::int32_t>> m_chg_log{{}}; 保存回溯信息。

回溯信息用于恢复环境。

环境的功能就是通过标识符名称获得标识符对应的值，也就是通过字符串 x 获得 x 的值。这里重载了运算符 [] 用于实现 symtbl["x"] 这样的搜索。

insert() 功能是

  • 如果发现冲突，先检查是否已经有了回溯信息。
    - 如果有，那么视为赋值运算，只要修改环境。
    - 如果没有，视为新定义，覆盖原来定义并更新回溯信息。
  • 如果没有冲突，插入环境中。
class SymbolTable {
private:
    // 环境信息
    unordered_map<std::string, std::int32_t> m_memory;

    // 回溯信息
    std::vector<std::unordered_map<std::string, std::int32_t>> m_chg_log{{}};
public:
    void insert(std::string _n, std::int32_t _v) {
        auto x = m_memory.find(_n);
        if (x != m_memory.end()) {
            auto y = m_chg_log.back().find(_n);
            if (y == m_chg_log.back().end())
                m_chg_log.back().emplace(x->first, x->second);
            x->second = _v;
        } else {
            m_memory.emplace(_n, _v);
        }
    }

    // 用于增加回溯信息。回溯信息保存使用了栈。
    void enterLet() {
        m_chg_log.push_back({});
    }

    // 用于恢复环境，使用最新的回溯信息恢复环境。
    // 因为题目特殊性，这里没有删除新定义的信息。
    void leaveLet() {
        for (auto &x: m_chg_log.back()) {
            m_memory[x.first] = x.second;
        }
        m_chg_log.pop_back();
    }

    std::int32_t operator [] (std::string &s) {
        return m_memory[s];
    }

    std::int32_t operator [] (std::string &&s) {
        return m_memory[s];
    }
};

接下来是语法分析的递归下降子程序法：

实现代码：


class Parser {
private:
    SymbolTable symtbl;
    std::string code;
    std::size_t sptr;

    bool skip(std::string &&v) {
        if (code.substr(sptr, v.size()) == v) {
            sptr += v.size();
            return true;
        }
        return false;
    }

    void WS() {
        if (std::isblank(code[sptr])) {
            ++sptr;
        }
    }

    std::string read_id() {
        std::string res;
        while (std::isalnum(code[sptr])) {
            res += code[sptr];
            ++sptr;
        }
        return res;
    }

    std::int32_t read_num() {
        std::string res;
        std::int32_t sign = 1;
        if (code[sptr] == '-') {
            sign = -sign;
            ++sptr;
        }
        while (std::isdigit(code[sptr])) {
            res += code[sptr];
            ++sptr;
        }
        return sign * std::stoi(res);
    }

public:
    Parser(std::string &&s): code(s), sptr(0) { }

    std::int32_t init() {
        return expr();
    }

    std::int32_t assign() {
        auto &&x = read_id();
        WS();
        auto &&res = expr();
        WS();
        symtbl.insert(x, res);
        while (true) {
            if (std::isalpha(code[sptr])) {
                x = read_id();
                WS();
                if (skip(")")) {
                    return symtbl[x];
                }
                res = expr();
                WS();
                symtbl.insert(x, res);
            } else {
                return expr();
            }
        }
        return 0;
    }

    std::int32_t expr() {
        if (skip("(")) {
            auto res = base();
            skip(")");
            return res;
        } else {
            if (std::isdigit(code[sptr]) || code[sptr] == '-') {
                return read_num();
            } else {
                return symtbl[read_id()];
            }
        }
        return 0;
    }

    std::int32_t base() {
        if (skip("let")) {
            WS();
            symtbl.enterLet();
            auto res = assign();
            symtbl.leaveLet();
            return res;
        } else if (skip("add")) {
            WS();
            return add();
        } else if (skip("mult")) {
            WS();
            return mult();
        }
        return 0;
    }

    std::int32_t add() {
        auto x = expr();
        WS();
        auto y = expr();
        return x + y;
    }

    std::int32_t mult() {
        auto x = expr();
        WS();
        auto y = expr();
        return x * y;
    }
};

最终，在 Solution 中调用：

class Solution {
public:
    int evaluate(string expression) {
        Parser parser(std::move(expression));
        return parser.init();
    }
};

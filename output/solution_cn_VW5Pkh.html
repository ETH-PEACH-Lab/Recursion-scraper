约瑟夫环其实是个周期函数
周期内是个一次函数, 斜率为k
只用找到人数n所在的那个周期以及周期开始的点(a, b)
返回b + k * (n - a)
就是答案了

比如人数为100, 报数为2
它在64, 128的周期内
周期开始的点为(64, 1)
(100 - 64) * 2 + 1 = 73

class Solution {
public int lastRemaining(int n, int k) {
return ysf(n, k) - 1;
}

public int ysf(int n, int k) {
    if (k == 1) {
        return n;
    }
    int a, b, x, y;
    a = 1;
    b = 1;
    while (true) {
        x = (a - b) - (k - 1);
        if (x < 0) {
            if (a == n) {
                break;
            }
            a = a + 1;
            b = b + k;
            b = b % a;
            if (b == 0) {
                b = a;
            }
        } else {
            x = (a - b) / (k - 1);
            if (n >= a && n < (a + x + 1)) {
                x = n - a;
                b = b + k * x;
                break;
            }
            y = (a - b) % (k - 1);
            a = a + x + 1;
            b = k - 1 - y;
            if (b == 0) {
                b = a;
            }
        }
    }
    return b;
}

}

import java.math.BigInteger;

class Solution {
    public int lastRemaining(int n, int k) {
        BigInteger a = YSF(BigInteger.valueOf(n), BigInteger.valueOf(k));
        int b = a.intValue() - 1;
        return b;
    }

    public BigInteger YSF(BigInteger n, BigInteger k) {
        if (k.compareTo(BigInteger.valueOf(1)) == 0) {
            return n;
        }
        BigInteger a, b, x, y;
        a = BigInteger.valueOf(1);
        b = BigInteger.valueOf(1);
        while (true) {
            x = (a.subtract(b)).subtract(k.subtract(BigInteger.valueOf(1)));
            if (x.compareTo(BigInteger.valueOf(0)) < 0) {
                if (a.compareTo(n) == 0) {
                    break;
                }
                a = a.add(BigInteger.valueOf(1));
                b = b.add(k);
                b = b.remainder(a);
                if (b.compareTo(BigInteger.valueOf(0)) == 0) {
                    b = a;
                }
            } else {
                x = (a.subtract(b)).divide(k.subtract(BigInteger.valueOf(1)));
                if (n.compareTo(a) >= 0 && n.compareTo(a.add(x).add(BigInteger.valueOf(1))) < 0) {
                    x = n.subtract(a);
                    b = b.add(k.multiply(x));
                    break;
                }
                y = (a.subtract(b)).remainder(k.subtract(BigInteger.valueOf(1)));
                a = a.add(x).add(BigInteger.valueOf(1));
                b = k.subtract(BigInteger.valueOf(1)).subtract(y);
            }
        }
        return b;
    }
}

class Solution {

        int b = a.intValue();

    }

    public BigInteger YSF(BigInteger n, BigInteger k) {
        if (k.compareTo(BigInteger.valueOf(1)) == 0) {
            return n.subtract(BigInteger.valueOf(1));
        }
        BigInteger a, b, c;
        a = BigInteger.valueOf(2);
        b = BigInteger.valueOf(0);
        while (a.compareTo(n) <= 0) {
            if ((b.add(k)).compareTo(a) >= 0) {
                b = (b.add(k)).remainder(a);
                a = a.add(BigInteger.valueOf(1));
                continue;
            }
            c = ((a.subtract(b)).subtract(BigInteger.valueOf(2))).divide((k.subtract(BigInteger.valueOf(1))));
            if ((a.add(c)).compareTo(n) > 0) {
                b = b.add(((n.subtract(a)).add(BigInteger.valueOf(1))).multiply(k));
                break;
            }
            a = a.add(c);
            b = b.add(c.multiply(k));
        }
        return b;
    }

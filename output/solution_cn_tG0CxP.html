class Solution {
public static void reorderList(ListNode head) {
ListNode mid = getMid(head);
ListNode tail = mid.next;
mid.next = null;
solve(head, reverse(tail));
}

public static ListNode getMid(ListNode node) {
    ListNode slow = node;
    ListNode fast = node;
    while (fast != null && fast.next != null) {
        slow = slow.next;
        fast = fast.next.next;
    }
    return slow;
}

public static ListNode reverse(ListNode head) {
    if (head == null || head.next == null) {
        return head;
    }
    ListNode two = head.next;
    ListNode newHead = reverse(two);
    two.next = head;
    head.next = null;
    return newHead;
}

public static ListNode solve(ListNode first, ListNode last) {
    if (first == null || last == null) {
        return first;
    }
    ListNode mid = solve(first.next, last.next);
    first.next = last;
    last.next = mid;
    return first;
}

}

class Solution {
public:
    void reorderList(ListNode* head) {
        ListNode* mid = getMid(head);
        ListNode* tail = mid->next;
        mid->next = NULL;
        head = solve(head, reverse(tail));
    }

    ListNode* getMid(ListNode* node) {
        ListNode* slow = node;
        ListNode* fast = node;
        while (fast != NULL && fast->next != NULL) {
            slow = slow->next;
            fast = fast->next->next;
        }
        return slow;
    }

    ListNode* reverse(ListNode* head) {
        if (head == NULL || head->next == NULL) {
            return head;
        }
        ListNode* two = head->next;
        ListNode* newHead = reverse(two);
        two->next = head;
        head->next = NULL;
        return newHead;
    }

    ListNode* solve(ListNode* first, ListNode* last) {
        if (first == NULL || last == NULL) {
            return first;
        }
        ListNode* mid = solve(first->next, last->next);
        first->next = last;
        last->next = mid;
        return first;
    }

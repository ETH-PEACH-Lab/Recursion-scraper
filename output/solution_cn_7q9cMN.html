方法一：迭代

⚠部分语言需要注意，当 $n=-2^{31}$ 时，$-n=2^{31}$ 比 $32$ 位整数的最大值还大，溢出了。可以转成 $64$ 位整数解决。

```py [sol-Python3]
class Solution:
def myPow(self, x: float, n: int) -> float:
ans = 1
if n < 0: # x^-n = (1/x)^n
n = -n
x = 1 / x
while n: # 从低到高枚举 n 的每个比特位
if n & 1: # 这个比特位是 1
ans = x # 把 x 乘到 ans 中
x = x # x 自身平方
n >>= 1 # 继续枚举下一个比特位
return ans


```java [sol-Java]
class Solution {
    public double myPow(double x, int N) {
        double ans = 1;
        long n = N;
        if (n < 0) { // x^-n = (1/x)^n
            n = -n;
            x = 1 / x;
        }
        while (n != 0) { // 从低到高枚举 n 的每个比特位
            if ((n & 1) == 1) { // 这个比特位是 1
                ans *= x; // 把 x 乘到 ans 中
            }
            x *= x; // x 自身平方
            n >>= 1; // 继续枚举下一个比特位
        }
        return ans;
    }
}

## 方法二：递归

当 $n<0$ 时，$x^n = (x^{-1})^{-n} = \left(\dfrac{1}{x}\right)^{-n}$。

当 $n\ge 0$ 时，我们有


x^n =
\begin{cases}
(x^{n/2})^2, & n\bmod 2 = 0     \\
(x^{\lfloor n/2\rfloor})^2\cdot x, & n\bmod 2 = 1     \\
\end{cases}


根据上式递归计算 $x^n$。

递归边界：$x^0 = 1$。

```py [sol-Python3]
class Solution:
    def myPow(self, x: float, n: int) -> float:
        if n < 0:
            return self.myPow(1 / x, -n)
        if n == 0:
            return 1.0
        return self.myPow(x, n // 2) ** 2 * (x if n % 2 else 1.0)

```java [sol-Java]
class Solution {
public double myPow(double x, int n) {
if (n < 0) {
return pow(1 / x, -(long) n);
}
return pow(x, n);
}

private double pow(double x, long n) {
    if (n == 0) {
        return 1;
    }
    double res = pow(x, n / 2);
    res *= res;
    if (n % 2 != 0) {
        res *= x;
    }
    return res;
}

}

```rust [sol-Rust]
impl Solution {
    fn pow(x: f64, n: i64) -> f64 {
        if n == 0 {
            return 1.0;
        }
        let mut res = Self::pow(x, n / 2);
        res *= res;
        if n % 2 != 0 {
            res *= x;
        }
        res
    }

    pub fn my_pow(x: f64, n: i32) -> f64 {
        if n < 0 {
            Self::pow(1.0 / x, -(n as i64))
        } else {
            Self::pow(x, n as i64)
        }
    }
}

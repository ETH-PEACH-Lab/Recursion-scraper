数位dp，递归
👤 liekkassi 👍 2 2022-03-27 🖼 ↗ 查看原 Solution

实质就是换一种暴力枚举的方式，保存了枚举的结果。代码适用于各个数字出现的次数。

class Solution {
    int dp[]=new int[]{0,1,20,300,4000,50000,600000,7000000,80000000,900000000};
        //通过一个简单的找规律，dp[i]=dp[i-1]+10^(i-1) i是位数
    public int countDigitOne(int n) {
        if(n<10)
            return n==0?0:1;
        String num=String.valueOf(n);
        int len=num.length()-1;// 位数-1
        int firstNum=num.charAt(0)-'0';//最高位的数字

        int countFirst_1=(int)Math.pow(10,len);//最高位是1的总数。
        //比如：10-99中：countFirst_1=count（10-19）=10； 100-999中：count（100-199）=100
        int nextNum=n-firstNum*countFirst_1; //把最高位除去。递归。
        int countsecond_1=dp[len]*firstNum;//dp[len]表示除去最高位的含有1的数。
        //dp[len]*firstNum 代表高位可取0~firstNum，非高位有dp[len]个1.
        //例如:33：dp[1]=1,拆分成1-30和3.1-30中末尾为1前面有三个可取的数。01，11，21

        //分成两种情况，高位为1和高位>1. countDigitOne(nextNum)+countsecond_1不变，变的是后面的
        if(firstNum==1){
            return  countDigitOne(nextNum)+countsecond_1+nextNum+1;
            //最后+1+nextnum有点难以理解，+1是因为0的存在，这个1就是加的仅仅只有最高位是1的情况
            //比如num=123，加的“100”这种情况，num=13,就是加的“10”这种情况。
            //加上nextnum是因为最高位是1，有多少个数，就有多少个1，
            //比如13，除了3中有一个1，还有11（第一个1），12，13 .nextnum=3，加1代表加上“10” countsecond_1代表“01”。countDigitOne(1)=1.（也就是11的第二个1））
            //加起来就是1+1+3+1=6
        }
        else{
            return countDigitOne(nextNum)+countsecond_1+countFirst_1;
            //最后加的是countFirst_1，也就是加上高位是1的情况就可以了。
            //比如33 只用额外加上10-19这10数中的第一个1就行了。
            //countsecond_1代表“01”“11”“21”中的第二个1.
            //countDigitOne(3)代表“31”中的1 
            //一共是1+3+10=14
        }
    }
}

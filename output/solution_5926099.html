Intuition

The problem is about matching a string s with a pattern p where:
- . matches any single character.
- * matches zero or more occurrences of the preceding character.

We need to determine if the entire string matches the pattern, making this a dynamic programming (DP) problem where we evaluate subproblems (smaller parts of the string and pattern) to solve the larger problem.

The goal is to determine whether the entire string s can be fully matched by the pattern p. For example:

  • Input: s = "aa", p = "a" returns True because a means zero or more as, which can match "aa".
  • Input: s = "ab", p = "." returns True because . means zero or more of any character, matching "ab".

Approach

We can solve this using dynamic programming. We'll define a 2D table dp where dp[i][j] is True if the substring s[0..i-1] matches the pattern p[0..j-1]. Otherwise, it's False.

Key Observations:

  1. Direct Match: If the current characters in s and p match, or if p[j-1] is a . (which matches any character), then dp[i][j] = dp[i-1][j-1].

  2. Handling *:
    - * matches zero occurrences of the preceding character (ignore the preceding character and *), so dp[i][j] = dp[i][j-2].
    - If s[i-1] matches the character before * (or it's a .), then dp[i][j] can also be True by matching one or more occurrences of that character. So, dp[i][j] = dp[i-1][j].

Base Cases:

  • dp[0][0] = True because an empty string matches an empty pattern.
  • When p[j-1] == '*', it can match zero occurrences of the preceding element, so we initialize dp[0][j] by checking if the pattern can match an empty string (dp[0][j] = dp[0][j-2]).

Complexity

  • Time complexity: The solution involves filling up a DP table of size (m+1) x (n+1), where m is the length of the string s and n is the length of the pattern p. This results in a time complexity of O(m * n).

  • Space complexity: We use a 2D table dp of size (m+1) x (n+1), resulting in a space complexity of O(m * n).

class Solution {
public:
bool isMatch(string s, string p) {
int m = s.length(), n = p.length();

    // dp[i][j] will be true if s[0..i-1] matches p[0..j-1]
    vector<vector<bool>> dp(m + 1, vector<bool>(n + 1, false));

    // Base case: empty string matches empty pattern
    dp[0][0] = true;

    // Handle cases where pattern contains '*' which can match an empty string
    for (int j = 1; j <= n; ++j) {
        if (p[j - 1] == '*') {
            dp[0][j] = dp[0][j - 2];  // '*' can match zero occurrence of the preceding element
        }
    }

    for (int i = 1; i <= m; ++i) {
        for (int j = 1; j <= n; ++j) {
            if (p[j - 1] == s[i - 1] || p[j - 1] == '.') {
                dp[i][j] = dp[i - 1][j - 1];  // Characters match or '.' matches any single character
            } else if (p[j - 1] == '*') {
                // '*' can match zero occurrence of the previous character
                dp[i][j] = dp[i][j - 2];
                // Or '*' can match one or more occurrences of the previous character
                if (p[j - 2] == s[i - 1] || p[j - 2] == '.') {
                    dp[i][j] = dp[i][j] || dp[i - 1][j];
                }
            }
        }
    }

    return dp[m][n];
}

};

#include <stdbool.h>
#include <string.h>

bool isMatch(char* s, char* p) {
    int m = strlen(s);
    int n = strlen(p);

    // Create a 2D DP table with (m+1) x (n+1) dimensions
    bool dp[m+1][n+1];

    // Initialize the DP table
    memset(dp, false, sizeof(dp));

    // Base case: empty string matches empty pattern
    dp[0][0] = true;

    // Handle patterns like a*, a*b*, a*b*c*, etc. that can match an empty string
    for (int j = 1; j <= n; ++j) {
        if (p[j-1] == '*') {
            dp[0][j] = dp[0][j-2];  // '*' can match zero occurrence of the preceding element
        }
    }

    // Fill the DP table
    for (int i = 1; i <= m; ++i) {
        for (int j = 1; j <= n; ++j) {
            // Case 1: The characters are the same, or pattern character is '.'
            if (p[j-1] == s[i-1] || p[j-1] == '.') {
                dp[i][j] = dp[i-1][j-1];
            }
            // Case 2: Pattern contains a '*' wildcard
            else if (p[j-1] == '*') {
                // '*' acts as zero of the preceding character
                dp[i][j] = dp[i][j-2];

                // If preceding character in the pattern matches the current character in the string
                // or if the preceding character in the pattern is '.', '*' can match more
                if (p[j-2] == s[i-1] || p[j-2] == '.') {
                    dp[i][j] = dp[i][j] || dp[i-1][j];
                }
            }
        }
    }

    return dp[m][n];
}

public class Solution {
public bool IsMatch(string s, string p) {
// Call the recursive matching function starting from index 0 for both string and pattern
return IsMatchHelper(s, p, 0, 0);
}

private bool IsMatchHelper(string s, string p, int i, int j) {
    // If we've reached the end of the pattern, check if we're also at the end of the string
    if (j == p.Length) {
        return i == s.Length;
    }

    // Check if the current characters in s and p match, considering '.'
    bool currentMatch = (i < s.Length) && (p[j] == s[i] || p[j] == '.');

    // Check for the '*' wildcard, which refers to the preceding element
    if (j + 1 < p.Length && p[j + 1] == '*') {
        // There are two cases:
        // 1. '*' can match 0 occurrences of the preceding element, so skip it (j+2)
        // 2. If the current characters match, '*' can match at least one occurrence, so move i (i+1)
        return IsMatchHelper(s, p, i, j + 2) || (currentMatch && IsMatchHelper(s, p, i + 1, j));
    } else {
        // If there is no '*' wildcard, proceed with a simple match and move both i and j
        return currentMatch && IsMatchHelper(s, p, i + 1, j + 1);
    }
}

}

for (let i = 1; i <= s.length; i++) {
    for (let j = 1; j <= p.length; j++) {
        if (p[j - 1] === '.' || p[j - 1] === s[i - 1]) {
            // If characters match or pattern has '.', copy the diagonal result
            dp[i][j] = dp[i - 1][j - 1];
        } else if (p[j - 1] === '*') {
            // If pattern has '*', we consider two scenarios:
            // 1. Treat '*' as repeating zero times: dp[i][j-2]
            // 2. Treat '*' as repeating the character one or more times: dp[i-1][j]
            if (p[j - 2] === s[i - 1] || p[j - 2] === '.') {
                dp[i][j] = dp[i][j - 2] || dp[i - 1][j];
            } else {
                dp[i][j] = dp[i][j - 2];
            }
        }
    }
}

return dp[s.length][p.length];

};

function isMatch(s: string, p: string): boolean {
    // dp[i][j] will be true if s[0...i-1] matches p[0...j-1]
    const dp: boolean[][] = Array(s.length + 1).fill(null).map(() => Array(p.length + 1).fill(false));

    // Empty string matches empty pattern
    dp[0][0] = true;

    // Handle patterns like a*, a*b*, etc., where '*' can match zero occurrences of the preceding element
    for (let j = 2; j <= p.length; j++) {
        if (p[j - 1] === '*') {
            dp[0][j] = dp[0][j - 2];
        }
    }

    // Iterate over the string and the pattern
    for (let i = 1; i <= s.length; i++) {
        for (let j = 1; j <= p.length; j++) {
            if (p[j - 1] === '.' || p[j - 1] === s[i - 1]) {
                // Characters match or pattern has '.', so the current state depends on the previous state
                dp[i][j] = dp[i - 1][j - 1];
            } else if (p[j - 1] === '*') {
                // '*' means zero or more of the preceding element
                dp[i][j] = dp[i][j - 2]; // Zero occurrences of the preceding element
                if (p[j - 2] === '.' || p[j - 2] === s[i - 1]) {
                    // One or more occurrences of the preceding element
                    dp[i][j] = dp[i][j] || dp[i - 1][j];
                }
            }
        }
    }

    return dp[s.length][p.length];
}

    # Empty string matches with empty pattern
    dp[0][0] = True

    # Deals with patterns like a*, a*b*, a*b*c* which can match an empty string
    for j in range(1, len(p) + 1):
        if p[j - 1] == '*':
            dp[0][j] = dp[0][j - 2]

    # Fill the DP table
    for i in range(1, len(s) + 1):
        for j in range(1, len(p) + 1):
            if p[j - 1] == '.' or p[j - 1] == s[i - 1]:
                # If characters match or pattern has '.', carry forward the previous result
                dp[i][j] = dp[i - 1][j - 1]

                # Check zero occurrence case or match multiple times
                dp[i][j] = dp[i][j - 2] or (dp[i - 1][j] and (p[j - 2] == s[i - 1] or p[j - 2] == '.'))

    # The answer is whether the entire string s matches the entire pattern p
    return dp[-1][-1]

class Solution:
    def isMatch(self, s: str, p: str) -> bool:
        # Initialize a DP table with False values
        dp = [[False] * (len(p) + 1) for _ in range(len(s) + 1)]

        # Base case: empty string matches with empty pattern
        dp[0][0] = True

        # Handle patterns with '*' where it can match zero preceding element
        for j in range(2, len(p) + 1):
            if p[j - 1] == '*':
                dp[0][j] = dp[0][j - 2]

        # Fill the DP table
        for i in range(1, len(s) + 1):
            for j in range(1, len(p) + 1):
                if p[j - 1] == '.' or p[j - 1] == s[i - 1]:
                    # Match single character or '.'
                    dp[i][j] = dp[i - 1][j - 1]
                elif p[j - 1] == '*':
                    # Check two cases:
                    # 1. '*' means zero occurrences of the preceding element
                    # 2. '*' means one or more occurrences of the preceding element
                    dp[i][j] = dp[i][j - 2] or ((p[j - 2] == s[i - 1] or p[j - 2] == '.') and dp[i - 1][j])

        return dp[len(s)][len(p)]

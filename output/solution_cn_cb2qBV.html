解题思路

记忆化递归（就是要缓存递归产生的结果，避免重复计算）
1. allPossileFBT(n)：表示所有含n个结点的可能的真二叉树的列表
2. 递归策略是allPossileFBT(n) = {root->left = allPossileFBT(x),root->right = allPossileFBT(n - x - 1)}其中x是要枚举的
3. 核心思想：一个真二叉树，他的左子树和右子树也一定是真二叉树
4. 注意点：
- 1.真二叉树的节点个数不可能是偶数，一定是奇数
- 2.为了避免重复计算，我们需要把递归产生的结果，也就是含有x个节点的真二叉树的列表保存起来，后面要用

代码

/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    //用于存储中间结果
    unordered_map<int,vector<TreeNode*>> mp;

    vector<TreeNode*> allPossibleFBT(int n) {
        /*记忆化递归（就是要缓存递归产生的结果，避免重复计算）
            allPossileFBT(n)：表示所有含n个结点的可能的真二叉树的列表
            递归策略是allPossileFBT(n) = {root->left = allPossileFBT(x),root->right = allPossileFBT(n - x - 1)}其中x是要枚举的
            核心思想：一个真二叉树，他的左子树和右子树也一定是真二叉树
            注意点：
                1.真二叉树的节点个数不可能是偶数，一定是奇数
                2.为了避免重复计算，我们需要把递归产生的结果，也就是含有x个节点的真二叉树的列表保存起来，后面要用
        */
        //如果节点数为n的结果我们已经做过并存起来了，就不重复计算了，直接跳过即可
        if(!mp.count(n)){
            vector<TreeNode*> vecTree;
            if(n == 1){
                vecTree.emplace_back(new TreeNode(0));
            }
            else if(n & 1){
                //枚举左子树的总结点数，因为左子树也一定是真二叉树，所以我们枚举奇数就行了
                for(int leftSum = 1;leftSum <= n-2;leftSum += 2){
                    int rightSum = n - leftSum - 1;
                    //下面的过程就是乘法原理，固定左右子树各自的数量之后，分别组合即可
                    for(auto&& leftNode : allPossibleFBT(leftSum)){
                        for(auto&& rightNode : allPossibleFBT(rightSum)){
                            TreeNode* root = new TreeNode(0);
                            root->left = leftNode;
                            root->right = rightNode;
                            vecTree.emplace_back(root);
                        }
                    }
                }
            }
            mp[n] = vecTree;
        }
        return mp[n];
    }
};

补充一个动态规划解法，思路其实和记忆化递归一摸一样

/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<TreeNode*> allPossibleFBT(int n) {
        /*动态规划（总体思路和记忆化递归类似）
            dp[i]代表节点个数为i个的真二叉树列表
            状态转移方程：
                类似于乘法原理，左边x个，右边n-x-1个，然后组合
            初始化：dp[1] = {new TreeNode(0)};
            遍历顺序：dp[i]用到的一定是比i小的dp数组，所以自然是顺序遍历
        */
        vector<vector<TreeNode*>> dp(n+1);
        if(!(n&1)){
            return {};
        }
        dp[1].emplace_back(new TreeNode(0));
        for(int i = 3;i <= n;++i){
            //枚举左子树个数，右子树个数就自然确定了
            for(int leftSum = 1;leftSum <= i-2;leftSum += 2){
                int rightSum = i - leftSum - 1;
                //乘法原理，进行组合即可
                for(auto&& leftNode : dp[leftSum]){
                    for(auto&& rightNode : dp[rightSum]){
                        TreeNode* root = new TreeNode(0);
                        root->left = leftNode;
                        root->right = rightNode;
                        dp[i].emplace_back(root);
                    }
                }
            }
        }
        return dp[n];
    }
};

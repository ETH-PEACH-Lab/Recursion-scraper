Rust/Go/Java 队列 or DP100%
👤 kvicii 👍 1 2022-05-04 🖼 ↗ 查看原 Solution

解题思路

  1. 常规方法是通过队列，每次出队1 ~ k - 1范围内的元素，同时添加到队尾，再将最后一个元素删除，循环这个过程直到队列中只有一个元素时，返回即可。
  2. 也可以通过DP的方式进行递归及迭代，假设DP方程为：dp[i] = j，i表示环的元素个数，j表示i个环的元素中最后剩下的元素。
    dp[1] = 0
dp[2] = (dp[1] + k) % 2
...
dp[n] = (dp[n - 1] + k) % n
  • 常规队列解

```Rust []
impl Solution {
pub fn find_the_winner(n: i32, k: i32) -> i32 {
use std::collections::VecDeque;
let mut queue = VecDeque::new();
for i in 1..=n {
queue.push_back(i);
}
while queue.len() > 1 {
for i in 1..k {
let first = queue.pop_front().unwrap();
queue.push_back(first);
}
queue.pop_front();
}
queue.pop_front().unwrap()
}
}

```Go []
func findTheWinner(n int, k int) int {
    queue := make([]int, n)
    for i := 1; i <= n; i++ {
        queue[i-1] = i
    }
    for len(queue) > 1 {
        for i := 1; i < k; i++ {
            first := queue[0]
            queue = queue[1:]
            queue = append(queue, first)
        }
        queue = queue[1:]
    }
    return queue[0]
}

```Java []
class Solution {
public int findTheWinner(int n, int k) {
Queue queue = new LinkedList<>();
for (int i = 1; i <= n; i++) {
queue.offer(i);
}

    while (queue.size() > 1) {
        for (int i = 0; i < k - 1; i++) {
            queue.offer(queue.remove());
        }
        queue.remove();
    }
    return queue.remove();
}

impl Solution {
    pub fn find_the_winner(n: i32, k: i32) -> i32 {

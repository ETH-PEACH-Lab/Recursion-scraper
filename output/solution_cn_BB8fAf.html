❤指针的角色：负责推进的，负责尾随的
👤 xiao_ben_zhu 👍 5 2020-05-15 🖼 ↗ 查看原 Solution

1.迭代法

  • 思路：

    开始让 null 作为上一节点，从头结点开始，顺着链表，让当前节点的 next 指针指向上一节点，直到走到链表尾。整个链表就反向了。


Time: O(n)
Space: O(1) 

var reverseList = function (head) {
  let prev = null       // 尾随cur的prev指针，开始时指向null
  let cur = head        // 推进指针，开始时指向头结点
  while (cur) {         // cur指针推进到null节点，则退出循环
    let next = cur.next // 暂存cur的下一节点
    cur.next = prev     // 将cur的next指针指向prev
    prev = cur          // 将prev更新为cur节点
    cur = next          // 将cur指针推进一个节点
  }                     // 退出while时，cur指向null，prev指向原链的尾节点
  return prev
};

2.尾递归

  • 什么是尾递归
    一个函数中递归的调用都出现在函数的末尾，则这个递归函数是尾递归的
  • 其实是迭代法的简化版：
    • 定义递归函数 ——【让cur的next指针指向prev】
    • 把下面两句操作放入递归函数中
      prev = cur cur = next

Time: O(n)
Space: O(1) 

var reverseList = (head) => {
  function letCurPointToPrev(prev, cur) { // 参数prev和cur就是函数的私有变量
    if (!cur) return prev // cur推进到了null，结束递归，返回prev
    let next = cur.next // 暂存cur的下一节点
    cur.next = prev     // 让cur的next指向prev
    return letCurPointToPrev(cur, next) // 尾递归，cur推进到next，prev尾随cur
  }
  return letCurPointToPrev(null, head) // null,head分别成为prev,cur的初始值
}

3.借助辅助栈

  • 思路：
    辅助栈，从头到尾将节点值推入栈，头结点在栈底，尾节点在栈顶，出栈时，尾节点就先出栈，构建新的反向链表

Time: O(n)
Space: O(n) 

var reverseList = (head) => {
  let stack = []
  let dummyHead = new ListNode(0) // 虚拟头结点
  let prev = dummyHead // prev是推进指针，初始指向虚拟头结点
  while (head) { // 将原链的节点值逐个推入栈
    stack.push(head.val)
    head = head.next
  }
  while (stack.length) { // 栈空了就退出循环
    prev.next = new ListNode(stack.pop()) // 节点值出栈 创建节点 prev的next指向它
    prev = prev.next // 推进prev指针，指向新创建的节点
  }
  return dummyHead.next
}
